  AppDatabase.forTesting() : super(NativeDatabase.memory());
  AppDatabase.forTestingWithConnection(DatabaseConnection c) : super(c);

  @override
  int get schemaVersion => 3;  // 改：2 -> 3

  @override
  MigrationStrategy get migration {
    return MigrationStrategy(
      onCreate: (m) async {
        await m.createAll();
        await m.database.customStatement(
          'CREATE UNIQUE INDEX IF NOT EXISTS idx_books_file_hash ON books(file_hash)',
        );
        await m.database.customStatement(
          'CREATE INDEX IF NOT EXISTS idx_book_tags_tag ON book_tags(tag)',
        );
      },
      onUpgrade: (m, from, to) async {
        if (from < 2) {
          await m.database.customStatement(
            'CREATE UNIQUE INDEX IF NOT EXISTS idx_books_file_hash ON books(file_hash)',
          );
        }
        if (from < 3) {
          // 加列：drift 默认 SQLite 不支持 add column in-DSL，用 customStatement
          await m.database.customStatement(
            'ALTER TABLE books ADD COLUMN sort_order INTEGER',
          );
          await m.createTable(bookTags);
          await m.database.customStatement(
            'CREATE INDEX IF NOT EXISTS idx_book_tags_tag ON book_tags(tag)',
          );
        }
      },
      beforeOpen: (details) async {
        await customStatement('PRAGMA foreign_keys = ON;');
      },
    );
  }
}

// app/lib/data/local/daos/book_tags_dao.dart
import 'package:drift/drift.dart';

import '../database.dart';
import '../tables/book_tags.dart';

part 'book_tags_dao.g.dart';

@DriftAccessor(tables: [BookTags])
class BookTagsDao extends DatabaseAccessor<AppDatabase> with _$BookTagsDaoMixin {
  BookTagsDao(super.db);

  Future<List<String>> tagsOfBook(String bookId) async {
    final rows = await (select(bookTags)
          ..where((t) => t.bookId.equals(bookId))
          ..orderBy([(t) => OrderingTerm(expression: t.tag)]))
        .get();
    return rows.map((r) => r.tag).toList();
  }

  Future<Set<String>> allTags() async {
    final rows = await select(bookTags).get();
    return rows.map((r) => r.tag).toSet();
  }

  Future<List<String>> bookIdsWithTag(String tag) async {
    final rows = await (select(bookTags)..where((t) => t.tag.equals(tag))).get();
    return rows.map((r) => r.bookId).toList();
  }

  Future<void> addTag({required String bookId, required String tag}) async {
    await into(bookTags).insertOnConflictUpdate(
      BookTagsCompanion.insert(bookId: bookId, tag: tag),
    );
  }

  Future<void> removeTag({required String bookId, required String tag}) async {
    await (delete(bookTags)
          ..where((t) => t.bookId.equals(bookId) & t.tag.equals(tag)))
        .go();
  }

  Future<void> replaceTagsForBook({
    required String bookId,
    required List<String> tags,
  }) async {
    await db.transaction(() async {
      await (delete(bookTags)..where((t) => t.bookId.equals(bookId))).go();
      for (final tag in tags) {
        await into(bookTags).insert(
          BookTagsCompanion.insert(bookId: bookId, tag: tag),
        );
      }
    });
  }

  Stream<List<String>> watchTagsOfBook(String bookId) {
    return (select(bookTags)
          ..where((t) => t.bookId.equals(bookId))
          ..orderBy([(t) => OrderingTerm(expression: t.tag)]))
        .watch()
        .map((rows) => rows.map((r) => r.tag).toList());
  }
}
```

// app/lib/services/epub_cover_extractor.dart
import 'dart:io';
import 'dart:typed_data';

import 'package:epubx/epubx.dart';
import 'package:path/path.dart' as p;

/// 从 EPUB 字节流提取封面图片，写入 [coversDir] 并返回路径。
/// 失败（无封面 / 解析异常）返回 null，调用方降级到占位图。
class EpubCoverExtractor {
  final Directory coversDir;

  EpubCoverExtractor({required this.coversDir});

  Future<String?> extract({
    required String bookId,
    required Uint8List epubBytes,
  }) async {
    if (epubBytes.isEmpty) return null;
    try {
      final book = await EpubReader.readBook(epubBytes);
      final cover = book.CoverImage;
      if (cover == null) return null;
      final ext = _extensionFromContentType(cover.MediaType ?? 'image/png');
      final file = File(p.join(coversDir.path, '$bookId.$ext'));
      await file.writeAsBytes(cover.Content ?? Uint8List(0));
      return file.path;
    } catch (_) {
      return null;
    }
  }

  String _extensionFromContentType(String ct) {
    if (ct.contains('jpeg') || ct.contains('jpg')) return 'jpg';
    if (ct.contains('png')) return 'png';
    if (ct.contains('webp')) return 'webp';
    if (ct.contains('gif')) return 'gif';
    return 'png';
  }
}
```

> 注意：`epubx` 的 API 大小写需要 verify，常见是 `CoverImage`（PascalCase） 或 `coverImage`（lowerCamelCase）。implementer 在实现时若 import 错误需改成实际 API。

// app/lib/services/book_import_service.dart
import 'dart:io';

import 'package:crypto/crypto.dart';
import 'package:path/path.dart' as p;
import 'package:reader_engine_epub_view/reader_engine_epub_view.dart';
import 'package:uuid/uuid.dart';

import '../data/local/file_storage.dart';
import '../data/models/book.dart';
import '../data/repositories/book_repository.dart';

class ImportResult {
  final Book book;
  final bool isNew;
  const ImportResult({required this.book, required this.isNew});
}

class BookImportService {
  final BookRepository _bookRepository;
  final FileStorage _fileStorage;
  final Uuid _uuid;
  final EpubCoverExtractor _coverExtractor;    // 新

  BookImportService({
    required BookRepository bookRepository,
    required FileStorage fileStorage,
    required EpubCoverExtractor coverExtractor,    // 新
    Uuid? uuid,
  })  : _bookRepository = bookRepository,
        _fileStorage = fileStorage,
        _coverExtractor = coverExtractor,
        _uuid = uuid ?? const Uuid();

  Future<ImportResult> importFile(
    File source, {
    required BookFormat format,
  }) async {
    final bytes = await source.readAsBytes();
    final hash = sha1.convert(bytes).toString();
    final extension = format == BookFormat.epub ? 'epub' : 'pdf';

    final existing = await _bookRepository.findByHash(hash);
    if (existing != null) {
      return ImportResult(book: existing, isNew: false);
    }

    final stored = await _fileStorage.storeLibraryFile(
      source: source,
      fileHash: hash,
      extension: extension,
    );

    final fileName = p.basenameWithoutExtension(source.path);

    // 仅 EPUB 提取封面；失败 coverPath=null
    String? coverPath;
    if (format == BookFormat.epub) {
      coverPath = await _coverExtractor.extract(
        bookId: '', // 占位；下面 save 后再 update
        epubBytes: bytes,
      );
    }

    final bookId = _uuid.v4();
    if (coverPath != null) {
      // 重命名为 bookId（避免用 hash 作为封面 key）
      final ext = p.extension(coverPath);
      final newPath = p.join(_fileStorage.coversDir.path, '$bookId$ext');
      await File(coverPath).rename(newPath);
      coverPath = newPath;
    }

    final book = Book(
      id: bookId,
      title: fileName,
      author: null,
      format: format,
      filePath: stored.path,
      fileHash: hash,
      fileSizeBytes: bytes.length,
      coverPath: coverPath,
      addedAt: DateTime.now(),
      lastReadAt: null,
    );

    try {
      await _bookRepository.save(book);
      return ImportResult(book: book, isNew: true);
    } on Exception catch (e) {
      if (e.toString().contains('UNIQUE constraint failed')) {
        final fallback = await _bookRepository.findByHash(hash);
        if (fallback != null) {
          return ImportResult(book: fallback, isNew: false);
        }
      }
      rethrow;
    }
  }
}

// app/lib/features/library/providers/library_controller.dart
import 'dart:async';

import 'package:flutter_riverpod/flutter_riverpod.dart';

import '../../../core/providers.dart';
import '../../../data/local/daos/book_tags_dao.dart';
import '../../../data/repositories/book_repository.dart';
import '../../../data/repositories/settings_repository.dart';
import '../models/library_state.dart';
import '../models/sort_by.dart';
import '../models/view_mode.dart';

class LibraryController extends StateNotifier<LibraryState> {
  final BookRepository _bookRepository;
  final BookTagsDao _bookTagsDao;
  final SettingsRepository _settingsRepository;
  StreamSubscription<List<Book>>? _booksSub;
  Timer? _searchDebounce;

  static const _keyViewMode = 'library.viewMode';
  static const _keySortBy = 'library.sortBy';

  LibraryController({
    required BookRepository bookRepository,
    required BookTagsDao bookTagsDao,
    required SettingsRepository settingsRepository,
  })  : _bookRepository = bookRepository,
        _bookTagsDao = bookTagsDao,
        _settingsRepository = settingsRepository,
        super(const LibraryState()) {
    _hydrateFromSettings();
    _subscribeToBooks();
  }

  Future<void> _hydrateFromSettings() async {
    final viewModeStr = await _settingsRepository.get<String>(_keyViewMode, 'grid');
    final sortByStr = await _settingsRepository.get<String>(_keySortBy, 'recentlyRead');
    state = state.copyWith(
      viewMode: viewModeStr == 'list' ? ViewMode.list : ViewMode.grid,
      sortBy: SortBy.values.firstWhere(
        (s) => s.name == sortByStr,
        orElse: () => SortBy.recentlyRead,
      ),
    );
  }

  void _subscribeToBooks() {
    _booksSub = _bookRepository.watchAll().listen((books) async {
      final tags = await _bookTagsDao.allTags();
      if (!mounted) return;
      state = state.copyWith(books: books, allTags: tags);
    });
  }

  Future<void> setViewMode(ViewMode mode) async {
    state = state.copyWith(viewMode: mode);
    await _settingsRepository.set<String>(_keyViewMode, mode.name);
  }

  Future<void> setSortBy(SortBy sortBy) async {
    state = state.copyWith(sortBy: sortBy);
    await _settingsRepository.set<String>(_keySortBy, sortBy.name);
  }

  Future<void> setSearchQuery(String query) async {
    _searchDebounce?.cancel();
    _searchDebounce = Timer(const Duration(milliseconds: 200), () {
      if (!mounted) return;
      state = state.copyWith(searchQuery: query);
    });
  }

  Future<void> toggleTagFilter(String tag) async {
    final next = Set<String>.from(state.activeTagFilters);
    if (!next.add(tag)) next.remove(tag);
    state = state.copyWith(activeTagFilters: next);
  }

  void enterSelectionMode(String initialBookId) {
    state = state.copyWith(
      mode: LibraryMode.selecting,
      selectedIds: {initialBookId},
    );
  }

  void toggleSelection(String bookId) {
    final next = Set<String>.from(state.selectedIds);
    if (!next.add(bookId)) next.remove(bookId);
    state = state.copyWith(selectedIds: next);
  }

  void exitSelectionMode() {
    state = state.copyWith(
      mode: LibraryMode.idle,
      selectedIds: const {},
    );
  }

  Future<void> deleteBook(String bookId) async {
    await _bookRepository.delete(bookId);
  }

  Future<void> batchDelete() async {
    if (state.selectedIds.isEmpty) return;
    final ids = Set<String>.from(state.selectedIds);
    await _bookRepository.deleteBooks(ids);
    exitSelectionMode();
  }

  Future<void> setManualSortOrder(List<String> orderedBookIds) async {
    for (var i = 0; i < orderedBookIds.length; i++) {
      await _bookRepository.setSortOrder(
        bookId: orderedBookIds[i],
        sortOrder: i,
      );
    }
  }

  @override
  void dispose() {
    _searchDebounce?.cancel();
    _booksSub?.cancel();
    super.dispose();
  }
}

// app/lib/features/library/widgets/book_cover.dart
import 'dart:io';

import 'package:flutter/material.dart';

class BookCover extends StatelessWidget {
  final File? coverFile;
  final String title;
  final double aspectRatio;

  const BookCover({
    super.key,
    required this.coverFile,
    required this.title,
    this.aspectRatio = 0.7,
  });

  @override
  Widget build(BuildContext context) {
    if (coverFile == null) return _Placeholder(title: title, aspectRatio: aspectRatio);
    return AspectRatio(
      aspectRatio: aspectRatio,
      child: Image.file(
        coverFile!,
        fit: BoxFit.cover,
        errorBuilder: (_, __, ___) => _Placeholder(
          title: title,
          aspectRatio: aspectRatio,
        ),
      ),
    );
  }
}

class _Placeholder extends StatelessWidget {
  final String title;
  final double aspectRatio;
  const _Placeholder({required this.title, required this.aspectRatio});

  @override
  Widget build(BuildContext context) {
    final firstChar = title.isNotEmpty ? title.characters.first : '?';
    final color = _colorFromString(title);
    return AspectRatio(
      aspectRatio: aspectRatio,
      child: Container(
        color: color,
        alignment: Alignment.center,
        child: Text(
          firstChar,
          style: const TextStyle(
            fontSize: 48,
            fontWeight: FontWeight.bold,
            color: Colors.white,
          ),
        ),
      ),
    );
  }

  Color _colorFromString(String s) {
    final palette = [
      Colors.blue.shade300,
      Colors.green.shade300,
      Colors.purple.shade300,
      Colors.orange.shade300,
      Colors.pink.shade300,
      Colors.teal.shade300,
    ];
    return palette[s.hashCode % palette.length];
  }
}

import 'dart:io';

import 'package:flutter/material.dart';

import '../../../data/models/book.dart';
import 'book_cover.dart';

class BookCard extends StatelessWidget {
  final Book book;
  final File? coverFile;
  final bool isSelected;
  final bool isSelectionMode;
  final VoidCallback onTap;
  final VoidCallback onLongPress;

  const BookCard({
    super.key,
    required this.book,
    required this.coverFile,
    this.isSelected = false,
    this.isSelectionMode = false,
    required this.onTap,
    required this.onLongPress,
  });

  @override
  Widget build(BuildContext context) {
    return GestureDetector(
      onTap: onTap,
      onLongPress: onLongPress,
      child: Stack(
        children: [
          Column(
            crossAxisAlignment: CrossAxisAlignment.stretch,
            children: [
              Expanded(child: BookCover(coverFile: coverFile, title: book.title)),
              const SizedBox(height: 4),
              Text(
                book.title,
                maxLines: 2,
                overflow: TextOverflow.ellipsis,
                style: Theme.of(context).textTheme.bodySmall,
              ),
              if (book.author != null)
                Text(
                  book.author!,
                  maxLines: 1,
                  overflow: TextOverflow.ellipsis,
                  style: Theme.of(context).textTheme.labelSmall,
                ),
            ],
          ),
          if (isSelectionMode)
            Positioned(
              top: 4,
              right: 4,
              child: Checkbox(
                value: isSelected,
                onChanged: (_) => onTap(),
              ),
            ),
        ],
      ),
    );
  }
}

class LibraryAppBar extends StatelessWidget implements PreferredSizeWidget {
  final ViewMode viewMode;
  final SortBy sortBy;
  final Set<String> allTags;
  final Set<String> activeTagFilters;
  final ValueChanged<ViewMode> onViewModeChanged;
  final ValueChanged<SortBy> onSortChanged;
  final ValueChanged<String> onSearch;
  final ValueChanged<String> onTagFilterToggled;
  final VoidCallback onImportTap;
  final bool searching;
  final TextEditingController? searchController;

  const LibraryAppBar({
    super.key,
    required this.viewMode,
    required this.sortBy,
    required this.allTags,
    required this.activeTagFilters,
    required this.onViewModeChanged,
    required this.onSortChanged,
    required this.onSearch,
    required this.onTagFilterToggled,
    required this.onImportTap,
    this.searching = false,
    this.searchController,
  });

  @override
  Size get preferredSize => const Size.fromHeight(kToolbarHeight);

  @override
  Widget build(BuildContext context) {
    return AppBar(
      title: searching
          ? TextField(
              controller: searchController,
              autofocus: true,
              decoration: const InputDecoration(
                hintText: '搜索书名/作者',
                border: InputBorder.none,
              ),
              onChanged: onSearch,
            )
          : const Text('智阅'),
      actions: [
        IconButton(
          icon: const Icon(Icons.search),
          onPressed: () {}, // 由 parent 切换 searching 状态
          tooltip: '搜索',
        ),
        if (allTags.isNotEmpty)
          IconButton(
            icon: const Icon(Icons.label_outline),
            onPressed: () => _showTagDialog(context),
            tooltip: '标签筛选',
          ),
        PopupMenuButton<SortBy>(
          icon: const Icon(Icons.sort),
          onSelected: onSortChanged,
          itemBuilder: (_) => const [
            PopupMenuItem(value: SortBy.recentlyRead, child: Text('最近阅读')),
            PopupMenuItem(value: SortBy.addedDesc, child: Text('添加时间')),
            PopupMenuItem(value: SortBy.titleAsc, child: Text('书名')),
            PopupMenuItem(value: SortBy.authorAsc, child: Text('作者')),
            PopupMenuItem(value: SortBy.progressDesc, child: Text('阅读进度')),
            PopupMenuItem(value: SortBy.manual, child: Text('手动排序')),

                selected: selected,

// app/lib/features/library/widgets/book_grid.dart
import 'dart:io';

import 'package:flutter/material.dart';
import 'package:reorderables/reorderables.dart';

import '../../../data/models/book.dart';
import 'book_card.dart';

typedef CoverResolver = File? Function(String bookId);

class BookGrid extends StatelessWidget {
  final List<Book> books;
  final CoverResolver coverResolver;
  final ValueChanged<String> onTap;
  final ValueChanged<String> onLongPress;
  final bool isSelectionMode;
  final Set<String> selectedIds;
  final bool manualSortEnabled;
  final ValueChanged<List<String>> onReorder;

  const BookGrid({
    super.key,
    required this.books,
    required this.coverResolver,
    required this.onTap,
    required this.onLongPress,
    required this.isSelectionMode,
    required this.selectedIds,
    required this.manualSortEnabled,
    required this.onReorder,
  });

  @override
  Widget build(BuildContext context) {
    if (manualSortEnabled) {
      return ReorderableWrap(
        spacing: 16,
        runSpacing: 16,
        onReorder: (oldIdx, newIdx) {
          final ordered = List<Book>.of(books);
          final moved = ordered.removeAt(oldIdx);
          ordered.insert(newIdx > oldIdx ? newIdx - 1 : newIdx, moved);
          onReorder(ordered.map((b) => b.id).toList());
        },
        children: books.map((b) => _card(b, context)).toList(),
      );
    }
    return GridView.builder(
      padding: const EdgeInsets.all(16),
      gridDelegate: const SliverGridDelegateWithMaxCrossAxisExtent(
        maxCrossAxisExtent: 160,
        childAspectRatio: 0.55,
        crossAxisSpacing: 16,
        mainAxisSpacing: 16,
      ),
      itemCount: books.length,
      itemBuilder: (_, i) => _card(books[i], context),
    );
  }

  Widget _card(Book b, BuildContext context) {
    return BookCard(
      book: b,
      coverFile: coverResolver(b.id),
      isSelected: selectedIds.contains(b.id),
      isSelectionMode: isSelectionMode,
      onTap: () => onTap(b.id),
      onLongPress: () => onLongPress(b.id),
    );
  }
}

    required this.isSelectionMode,
    required this.selectedIds,
    required this.manualSortEnabled,
    required this.onReorder,
  });

  @override
  Widget build(BuildContext context) {
    final tiles = books.map((b) => BookListTile(
      book: b,
      coverFile: coverResolver(b.id),
      isSelected: selectedIds.contains(b.id),
      isSelectionMode: isSelectionMode,
      onTap: () => onTap(b.id),
      onLongPress: () => onLongPress(b.id),
    )).toList();

    if (manualSortEnabled) {
      return ReorderableColumn(
        onReorder: (oldIdx, newIdx) {
          final ordered = List<Book>.of(books);
          final moved = ordered.removeAt(oldIdx);
          ordered.insert(newIdx > oldIdx ? newIdx - 1 : newIdx, moved);
          onReorder(ordered.map((b) => b.id).toList());
        },
        children: tiles,
      );
    }
    return ListView(children: tiles);
  }

  Widget _buildBody(BuildContext context, LibraryState state) {

            ? EmptyLibraryMode.noSearchResult
            : EmptyLibraryMode.empty,
      );

      final async = ref.watch(coverLoaderProvider(id));

      if (state.mode == LibraryMode.selecting) {
        ref.read(libraryControllerProvider.notifier).toggleSelection(id);
      } else {

    final onLongPress = (String id) {

        ref.read(libraryControllerProvider.notifier).enterSelectionMode(id);

        ref.read(libraryControllerProvider.notifier).toggleSelection(id);
      }
    };
    final onContextMenu = (String id) async {
      final book = state.books.firstWhere((b) => b.id == id);
      final action = await showBookContextMenu(context, title: book.title);
      if (action == null) return;
      switch (action) {
        case BookContextMenuAction.open:
          if (context.mounted) context.push('/book/$id');

    if (result == null || result.files.single.path == null) return;
    final path = result.files.single.path!;
    final format = path.toLowerCase().endsWith('.pdf') ? BookFormat.pdf : BookFormat.epub;
    final importService = await ref.read(bookImportServiceProvider.future);
    final importResult = await importService.importFile(File(path), format: format);

          : '已在书库中：${importResult.book.title}')),
