```dart
import 'package:html/dom.dart';
import 'package:html/parser.dart';

/// A character span within [HtmlTextUtils.htmlToParagraphs] result,
/// identifying one block-level paragraph's location in `plainText`.
class ParagraphSpan {
  final int start;
  final int end;
  const ParagraphSpan(this.start, this.end);

  int get length => end - start;

  @override
  bool operator ==(Object other) =>
      other is ParagraphSpan && start == other.start && end == other.end;

  @override
  int get hashCode => Object.hash(start, end);
}

/// Pure utility for converting HTML strings to plain text.
class HtmlTextUtils {
  /// HTML tag names treated as block-level (paragraph boundaries).
  static const _blockTags = <String>{
    'p', 'div', 'h1', 'h2', 'h3', 'h4', 'h5', 'h6',
    'li', 'blockquote', 'section', 'article', 'header', 'footer', 'aside',
  };

  /// Convert [html] to plain text by walking all text nodes in document order.
  /// Strips `<script>` and `<style>` subtrees first.
  static String htmlToPlainText(String html) {
    if (html.isEmpty) return '';
    final doc = HtmlParser(html).parse();
    for (final e in doc.querySelectorAll('script, style')) {
      e.remove();
    }
    final buf = StringBuffer();
    _walkText(doc, buf);
    return buf.toString();
  }

  /// Convert [html] to plain text AND identify paragraph spans within it.
  ///
  /// The plainText output is identical to [htmlToPlainText] (concatenation of
  /// all text nodes in document order, including whitespace text nodes
  /// between block elements). Paragraph spans identify block-level chunks
  /// (each `<p>`, `<div>`, etc.) and exclude between-element whitespace.
  ///
  /// Used by [Paginator] so that page breaks and `currentChapterPlainText`
  /// share one coordinate space.
  static ({String plainText, List<ParagraphSpan> paragraphs}) htmlToParagraphs(
      String html) {
    if (html.isEmpty) return (plainText: '', paragraphs: const []);
    final doc = HtmlParser(html).parse();
    for (final e in doc.querySelectorAll('script, style')) {
      e.remove();
    }
    final buf = StringBuffer();
    final paragraphs = <ParagraphSpan>[];
    _walkForParagraphs(doc, buf, paragraphs, inBlock: false);
    return (plainText: buf.toString(), paragraphs: paragraphs);
  }

  static void _walkText(Node node, StringBuffer buf) {
    for (final n in node.nodes) {
      if (n.nodeType == Node.TEXT_NODE) {
        buf.write(n.text);
      } else {
        _walkText(n, buf);
      }
    }
  }

  static void _walkForParagraphs(
    Node node,
    StringBuffer buf,
    List<ParagraphSpan> paragraphs, {
    required bool inBlock,
  }) {
    for (final n in node.nodes) {
      if (n.nodeType == Node.TEXT_NODE) {
        final text = n.text ?? '';
        if (text.isEmpty) continue;
        if (inBlock) {
          buf.write(text);
        } else {
          final start = buf.length;
          buf.write(text);
          paragraphs.add(ParagraphSpan(start, buf.length));
        }
      } else if (n.nodeType == Node.ELEMENT_NODE) {
        final element = n as Element;
        final tag = element.localName?.toLowerCase();
        final isBlock = tag != null && _blockTags.contains(tag);
        if (isBlock && !inBlock) {
          // Top-level block element: open a new paragraph span.
          final start = buf.length;
          _walkForParagraphs(element, buf, paragraphs, inBlock: true);
          if (buf.length > start) {
            paragraphs.add(ParagraphSpan(start, buf.length));
          }
        } else {
          // Inline element OR nested block (we're already inBlock):
          // continue current context without opening a new span.
          _walkForParagraphs(element, buf, paragraphs, inBlock: inBlock);
        }
      }
    }
  }
}
```

```dart
import 'package:flutter/widgets.dart';
import 'package:reader_engine/core/typography_settings.dart';
import 'package:reader_engine/rendering/html_text_utils.dart';

/// Re-exported for backwards compatibility. T1 moves [PageBreak] to
/// `reader_engine/lib/core/page_break.dart`; this re-export will be removed.
export 'package:reader_engine/core/page_break.dart';
import 'package:reader_engine/core/page_break.dart';

/// TextPainter-driven chapter paginator.
///
/// Plain text extraction delegated to [HtmlTextUtils] so that page break
/// offsets share the same coordinate space as
/// `ReaderEngine.currentChapterPlainText`.
class Paginator {
  Paginator();

  final _textCache = <_CacheKey, ({String plainText, List<PageBreak> breaks})>{};

  List<PageBreak> paginate({
    required String chapterHtml,
    required Size viewport,
    required TypographySettings typography,
  }) => paginateWithText(
        chapterHtml: chapterHtml,
        viewport: viewport,
        typography: typography,
      ).breaks;

  ({String plainText, List<PageBreak> breaks}) paginateWithText({
    required String chapterHtml,
    required Size viewport,
    required TypographySettings typography,
  }) {
    final key = _CacheKey(chapterHtml, viewport, typography);
    return _textCache.putIfAbsent(
      key,
      () => _paginateUncached(chapterHtml, viewport, typography),
    );
  }

  ({String plainText, List<PageBreak> breaks}) _paginateUncached(
    String chapterHtml,
    Size viewport,
    TypographySettings typography,
  ) {
    final parsed = HtmlTextUtils.htmlToParagraphs(chapterHtml);
    final plainText = parsed.plainText;
    final paragraphs = parsed.paragraphs;

    if (plainText.isEmpty) {
      return (
        plainText: '',
        breaks: const [PageBreak(startOffset: 0, endOffset: 0)],
      );
    }

    final style = _textStyle(typography);
    final usableHeight = viewport.height;

    final breaks = <PageBreak>[];
    var pageStart = 0;
    var pageHeight = 0.0;

    for (final p in paragraphs) {
      if (p.length == 0) continue;
      final paraText = plainText.substring(p.start, p.end);
      final paraHeight = _measureHeight(paraText, style, viewport.width);
      if (pageHeight + paraHeight > usableHeight && pageHeight > 0) {
        breaks.add(PageBreak(startOffset: pageStart, endOffset: p.start));
        pageStart = p.start;
        pageHeight = 0;
      }
      if (pageHeight == 0) {
        pageStart = p.start;
      }
      pageHeight += paraHeight;
    }
    breaks.add(PageBreak(startOffset: pageStart, endOffset: plainText.length));
    return (plainText: plainText, breaks: breaks);
  }

  TextStyle _textStyle(TypographySettings t) {
    return TextStyle(fontSize: t.fontSize, height: t.lineHeight);
  }

  double _measureHeight(String text, TextStyle style, double maxWidth) {
    if (text.isEmpty) return 0;
    final tp = TextPainter(
      text: TextSpan(text: text, style: style),
      textDirection: TextDirection.ltr,
    )..layout(maxWidth: maxWidth);
    final h = tp.height;
    tp.dispose();
    return h;
  }
}

class _CacheKey {
  final String html;
  final Size viewport;
  final TypographySettings typography;
  const _CacheKey(this.html, this.viewport, this.typography);

  @override
  bool operator ==(Object other) =>
      other is _CacheKey &&
      html == other.html &&
      viewport == other.viewport &&
      typography == other.typography;

  @override
  int get hashCode => Object.hash(html, viewport, typography);
}
```

```dart
import 'package:flutter/material.dart';
import 'package:flutter/rendering.dart';
import 'package:reader_engine/core/page_break.dart';
import 'package:reader_engine/rendering/pagination_model.dart';

/// Scrolls a [Scrollable] to make the character at [targetOffset] visible.
///
/// Two modes:
/// - **scroll**: single [SingleChildScrollView] holding the whole chapter.
///   Collect all [RenderParagraph]s, accumulate text lengths to find owner,
///   compute caret via `getOffsetForCaret`, animate to that y.
/// - **paged**: [PageView] where each page is one viewport-sized [Text].
///   Find the page whose break range contains [targetOffset] and jumpToPage.
///   No in-page scroll (page IS viewport; target is naturally visible).
///
/// [targetOffset] is in the same plainText coordinate space as
/// `ReaderEngine.currentChapterPlainText` (post-T0 unification).
class ScrollToOffsetService {
  static Future<void> scrollToCharacterOffset({
    required BuildContext context,
    required ScrollableState scrollable,
    required int targetOffset,
    required PaginationMode paginationMode,
    List<PageBreak> pageBreaks = const [],
    PageController? pageController,
    Duration duration = const Duration(milliseconds: 300),
  }) async {
    if (paginationMode == PaginationMode.scroll) {
      await _scrollInScrollView(
        scrollable: scrollable,
        targetOffset: targetOffset,
        duration: duration,
      );
      return;
    }

    // paged mode
    if (pageController == null) {
      assert(false, 'pageController required in paged mode');
      return;
    }
    if (pageBreaks.isEmpty) return;

    final pageIdx = _findPageForOffset(pageBreaks, targetOffset);
    pageController.jumpToPage(pageIdx);
  }

  static int _findPageForOffset(List<PageBreak> breaks, int offset) {
    for (var i = 0; i < breaks.length; i++) {
      if (breaks[i].endOffset > offset) return i;
    }
    return breaks.length - 1;
  }

  static Future<void> _scrollInScrollView({
    required ScrollableState scrollable,
    required int targetOffset,
    required Duration duration,
  }) async {
    final scrollRenderObj = scrollable.context.findRenderObject();
    if (scrollRenderObj == null) return;

    final paragraphs = _collectRenderParagraphs(scrollRenderObj);
    if (paragraphs.isEmpty) return;

    int cumulative = 0;
    RenderParagraph? owner;
    int localOffset = 0;
    for (final p in paragraphs) {
      final len = p.text.toPlainText().length;
      if (cumulative + len > targetOffset) {
        owner = p;
        localOffset = targetOffset - cumulative;
        break;
      }
      cumulative += len;
    }
    if (owner == null) return;

    final caretLocal = owner.getOffsetForCaret(
      TextPosition(offset: localOffset),
      Rect.zero,
    );
    final transform = owner.getTransformTo(scrollRenderObj);
    final caretInScroll = MatrixUtils.transformPoint(transform, caretLocal);

    await scrollable.position.animateTo(
      caretInScroll.dy,
      duration: duration,
      curve: Curves.easeInOut,
    );
  }

  static List<RenderParagraph> _collectRenderParagraphs(RenderObject root) {
    final result = <RenderParagraph>[];
    void visit(RenderObject obj) {
      if (obj is RenderParagraph) result.add(obj);
      obj.visitChildren(visit);
    }
    visit(root);
    return result;
  }
}
```

```dart
import 'package:zhiyue/data/models/annotation.dart';
import 'package:zhiyue/data/models/book.dart';
import 'package:zhiyue/data/models/book_note.dart';

/// Pure-function Markdown export.
/// Same inputs → same output (no DateTime.now() calls; inject via [now]).
class MarkdownExporter {
  /// Render the entire [bookNote] (with [book] metadata) as Obsidian-compatible
  /// Markdown. Conditional sections omitted when source data is empty.
  static String export({
    required Book book,
    required BookNote bookNote,
    required DateTime now,
  }) {
    final annotations = bookNote.annotations.toList()
      ..sort((a, b) {
        final ai = _chapterIndexOf(a);
        final bi = _chapterIndexOf(b);
        if (ai != bi) return ai.compareTo(bi);
        return a.createdAt.compareTo(b.createdAt);
      });

    final highlights = annotations.where((a) => a.note == null || a.note!.isEmpty).length;
    final notes = annotations.length - highlights;
    final tags = _collectAllTags(annotations);

    final createdDates = annotations.map((a) => a.createdAt).toList()..sort();
    final updatedDates = annotations.map((a) => a.updatedAt).toList()..sort();
    final readingStarted = createdDates.isEmpty ? null : createdDates.first;
    final readingEnded = updatedDates.isEmpty ? null : updatedDates.last;

    final buf = StringBuffer();

    // Frontmatter
    buf.writeln('---');
    buf.writeln('title: "${_escapeYaml(book.title)}"');
    if (book.author != null && book.author!.isNotEmpty) {
      buf.writeln('author: "${_escapeYaml(book.author!)}"');
    }
    buf.writeln('source: "${_sourceLabel(book.format)}"');
    buf.writeln('exported_at: ${now.toIso8601String()}');
    buf.writeln('total_highlights: $highlights');
    buf.writeln('total_notes: $notes');
    buf.writeln('reading_started: ${readingStarted?.toIso8601String() ?? 'null'}');
    buf.writeln('reading_ended: ${readingEnded?.toIso8601String() ?? 'null'}');
    buf.writeln('tags: [${tags.map((t) => _escapeYaml(t)).join(', ')}]');
    buf.writeln('---');
    buf.writeln();

    // Title
    buf.writeln('# 《${book.title}》读书笔记');
    buf.writeln();

    // Reading summary line
    final startedStr = readingStarted == null
        ? '（暂无）'
        : _formatDate(readingStarted);
    final endedStr = readingEnded == null ? '（暂无）' : _formatDate(readingEnded);
    buf.writeln('> 阅读时间：$startedStr ~ $endedStr');
    buf.writeln('> 共 $highlights 条高亮 · $notes 条笔记');
    buf.writeln();

    // 核心摘要 (conditional)
    if (bookNote.summary.isNotEmpty) {
      buf.writeln('## 核心摘要');
      buf.writeln();
      buf.writeln(bookNote.summary);
      buf.writeln();
    }

    // 核心知识点 (conditional)
    if (bookNote.keyTakeaways.isNotEmpty) {
      buf.writeln('## 核心知识点');
      buf.writeln();
      for (final k in bookNote.keyTakeaways) {
        buf.writeln('- $k');
      }
      buf.writeln();
    }

    // 高亮与笔记
    buf.writeln('## 高亮与笔记');
    buf.writeln();

    if (annotations.isEmpty) {
      buf.writeln('（暂无高亮）');
      buf.writeln();
    } else {
      _groupByChapter(annotations).forEach((chapterKey, items) {
        buf.writeln('### $chapterKey');
        buf.writeln();
        for (final a in items) {
          buf.writeln('> ${_colorEmoji(a.color)} ${a.quote}');
          if (a.hasNote) {
            buf.writeln('- 💬 ${a.note}');
          }
          if (a.tags.isNotEmpty) {
            buf.writeln('- 🏷️ ${a.tags.map((t) => '#$t').join(' ')}');
          }
          buf.writeln();
        }
      });
    }

    buf.writeln('---');
    buf.writeln();

    // 标签 (conditional)
    if (tags.isNotEmpty) {
      buf.writeln('## 标签');
      buf.writeln();
      buf.writeln(tags.map((t) => '#$t').join(' '));
      buf.writeln();
    }

    return buf.toString();
  }

  /// Render a single annotation as a Markdown fragment (clipboard use).
  static String exportSingle(Annotation annotation, {String? chapterTitle}) {
    final buf = StringBuffer();
    buf.writeln('> ${_colorEmoji(annotation.color)} ${annotation.quote}');
    if (annotation.hasNote) {
      buf.writeln('- 💬 ${annotation.note}');
    }
    if (annotation.tags.isNotEmpty) {
      buf.writeln('- 🏷️ ${annotation.tags.map((t) => '#$t').join(' ')}');
    }
    if (chapterTitle != null) {
      buf.writeln('- 📖 $chapterTitle');
    }
    return buf.toString().trimRight();
  }

  // ---- helpers ----

  static int _chapterIndexOf(Annotation a) {
    final loc = a.locator;
    if (loc is EpubRangeLocator) return loc.chapterIndex;
    if (loc is EpubCfiLocator) return loc.chapterIndex;
    return 0;
  }

  static String _chapterKey(Annotation a) {
    final loc = a.locator;
    if (loc is EpubRangeLocator) {
      final title = loc.chapterTitle;
      if (title != null && title.isNotEmpty) return title;
      return '第 ${loc.chapterIndex + 1} 章';
    }
    if (loc is EpubCfiLocator) {
      final title = loc.chapterTitle;
      if (title != null && title.isNotEmpty) return title;
      return '第 ${loc.chapterIndex + 1} 章';
    }
    return '未分类';
  }

  static Map<String, List<Annotation>> _groupByChapter(
      List<Annotation> annotations) {
    final map = <String, List<Annotation>>{};
    for (final a in annotations) {
      map.putIfAbsent(_chapterKey(a), () => []).add(a);
    }
    return map;
  }

  static List<String> _collectAllTags(List<Annotation> annotations) {
    final set = <String>{};
    for (final a in annotations) {
      set.addAll(a.tags);
    }
    return set.toList()..sort();
  }

  static String _colorEmoji(HighlightColor c) {
    switch (c) {
      case HighlightColor.yellow: return '🟡';
      case HighlightColor.green:  return '🟢';
      case HighlightColor.blue:   return '🔵';
      case HighlightColor.red:    return '🔴';
      case HighlightColor.purple: return '🟣';
    }
  }

  static String _sourceLabel(BookFormat f) =>
      f == BookFormat.epub ? 'EPUB' : 'PDF';

  static String _formatDate(DateTime d) =>
      '${d.year.toString().padLeft(4, '0')}-'
      '${d.month.toString().padLeft(2, '0')}-'
      '${d.day.toString().padLeft(2, '0')}';

  static String _escapeYaml(String s) => s.replaceAll('"', r'\"');
}
```

```dart
import 'dart:io';

import 'package:file_picker/file_picker.dart';

class NoteFileWriter {
  /// Opens a directory picker (Windows/macOS/Linux via file_picker) and writes
  /// [content] to `<pickedDir>/<suggestedName>.md`. Returns the absolute path
  /// on success or null if user cancelled.
  ///
  /// If file exists, appends ` (1)`, ` (2)`, etc. before extension.
  static Future<String?> saveMarkdownFile({
    required String content,
    required String suggestedName,
  }) async {
    final dir = await FilePicker.platform.getDirectoryPath();
    if (dir == null) return null;

    var name = '$suggestedName.md';
    var path = '$dir${Platform.pathSeparator}$name';
    var counter = 1;
    while (await File(path).exists()) {
      name = '$suggestedName ($counter).md';
      path = '$dir${Platform.pathSeparator}$name';
      counter++;
    }

    await File(path).writeAsString(content);
    return path;
  }

  /// Replace filesystem-forbidden chars + collapse whitespace + truncate.
  static String sanitizeForFilename(String raw) {
    var s = raw.replaceAll(RegExp(r'[\\/:*?"<>|]'), '_');
    s = s.trim().replaceAll(RegExp(r'\s+'), ' ');
    if (s.length > 80) s = s.substring(0, 80);
    return s;
  }
}
```

- [ ] **Step 5: Create `notes_page.dart`**

```dart
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:go_router/go_router.dart';

import '../../../core/locator.dart';
import '../../../data/models/annotation.dart';
import '../../../data/models/book_note.dart';
import '../../reader/widgets/annotation_tile.dart';
import '../../reader/widgets/note_editor_sheet.dart';
import '../providers/notes_page_providers.dart';
import '../services/markdown_exporter.dart';
import '../services/note_file_writer.dart';
import '../widgets/annotation_context_sheet.dart';

class NotesPage extends ConsumerWidget {
  final String bookId;

  const NotesPage({super.key, required this.bookId});

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final annotationsAsync = ref.watch(notesPageAnnotationsProvider(bookId));
    final noteAsync = ref.watch(notesPageBookNoteProvider(bookId));

    return Scaffold(
      appBar: AppBar(
        title: noteAsync.maybeWhen(
          data: (n) => Text(n == null ? '笔记' : '《${n.title.replaceFirst('《', '').replaceFirst('》读书笔记', '')}》笔记'),
          orElse: () => const Text('笔记'),
        ),
        actions: [
          IconButton(
            icon: const Icon(Icons.download),
            tooltip: '导出 Markdown',
            onPressed: () => _export(context, ref),
          ),
        ],
      ),
      body: annotationsAsync.when(
        loading: () => const Center(child: CircularProgressIndicator()),
        error: (e, _) => Center(
          child: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              Text('加载失败：$e'),
              TextButton(
                onPressed: () => ref.invalidate(notesPageAnnotationsProvider(bookId)),
                child: const Text('重试'),
              ),
            ],
          ),
        ),
        data: (annotations) => _body(context, ref, annotations),
      ),
    );
  }

  Widget _body(BuildContext context, WidgetRef ref, List<Annotation> annotations) {
    if (annotations.isEmpty) {
      return Center(
        child: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            const Text('本书暂无高亮与笔记'),
            const SizedBox(height: 12),
            FilledButton(
              onPressed: () => context.push('/book/$bookId'),
              child: const Text('去阅读器添加'),
            ),
          ],
        ),
      );
    }

    final highlights = annotations.where((a) => !a.hasNote).length;
    final notes = annotations.length - highlights;

    final groups = <int, List<Annotation>>{};
    for (final a in annotations) {
      final idx = _chapterIndex(a);
      groups.putIfAbsent(idx, () => []).add(a);
    }
    final sortedKeys = groups.keys.toList()..sort();

    return ListView(
      children: [
        Padding(
          padding: const EdgeInsets.all(16),
          child: Text(
            '共 $highlights 条高亮 · $notes 条笔记 · 更新于 '
            '${_latestDate(annotations)}',
            style: Theme.of(context).textTheme.bodySmall,
          ),
        ),
        for (final chapterIdx in sortedKeys) ...[
          ListTile(
            title: Text(
              _chapterTitle(groups[chapterIdx]!.first),
              style: const TextStyle(fontWeight: FontWeight.bold),
            ),
          ),
          for (final a in groups[chapterIdx]!)
            AnnotationTile(
              annotation: a,
              showMeta: true,
              onTap: () => context.push('/book/$bookId', extra: a.locator),
              onLongPress: () => _showContextMenu(context, ref, a),
            ),
        ],
      ],
    );
  }

  Future<void> _showContextMenu(
      BuildContext context, WidgetRef ref, Annotation a) async {
    final action = await showAnnotationContextSheet(context, a,
        chapterTitle: _chapterTitle(a));
    if (action == null) return;
    switch (action) {
      case AnnotationContextAction.edit:
        if (!context.mounted) return;
        await showModalBottomSheet<void>(
          context: context,
          isScrollControlled: true,
          builder: (_) => NoteEditorSheet(
            annotation: a,
            suggestions: const [],
            onSave: (note, color) async {
              await ref.read(notesRepositoryOverride).updateAnnotation(
                  a.copyWith(note: note, color: color, updatedAt: DateTime.now()));
            },
            onDelete: () async {
              await ref.read(notesRepositoryOverride).deleteAnnotation(a.id);
            },
          ),
        );
        break;
      case AnnotationContextAction.delete:
        await ref.read(notesRepositoryOverride).deleteAnnotation(a.id);
        break;
      case AnnotationContextAction.copyQuote:
      case AnnotationContextAction.copyMarkdown:
        // Clipboard already set inside the sheet.
        break;
    }
  }

  Future<void> _export(BuildContext context, WidgetRef ref) async {
    final note = await ref.read(notesPageBookNoteProvider(bookId)).maybeWhen(
          data: (n) => n,
          orElse: () => null,
        );
    if (note == null || !context.mounted) return;
    final markdown = MarkdownExporter.export(
      bookNote: note,
      book: _stubBookForExport(note),  // M3.4 simplification: use note.title
      now: DateTime.now(),
    );
    final name = NoteFileWriter.sanitizeForFilename(note.title) + ' - 读书笔记';
    final path = await NoteFileWriter.saveMarkdownFile(
      content: markdown,
      suggestedName: name,
    );
    if (!context.mounted) return;
    if (path == null) return;  // user cancelled, silent
    // SnackBar handled by T8 (open-folder action)
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(content: Text('已保存：$path')),
    );
  }

  // Local helpers
  int _chapterIndex(Annotation a) {
    final loc = a.locator;
    if (loc is EpubRangeLocator) return loc.chapterIndex;
    if (loc is EpubCfiLocator) return loc.chapterIndex;
    return 0;
  }

  String _chapterTitle(Annotation a) {
    final loc = a.locator;
    if (loc is EpubRangeLocator) {
      return loc.chapterTitle?.isNotEmpty == true
          ? loc.chapterTitle!
          : '第 ${loc.chapterIndex + 1} 章';
    }
    if (loc is EpubCfiLocator) {
      return loc.chapterTitle?.isNotEmpty == true
          ? loc.chapterTitle!
          : '第 ${loc.chapterIndex + 1} 章';
    }
    return '未分类';
  }

  String _latestDate(List<Annotation> annotations) {
    final latest = annotations.map((a) => a.updatedAt).reduce((a, b) => a.isAfter(b) ? a : b);
    return '${latest.month.toString().padLeft(2, '0')}-${latest.day.toString().padLeft(2, '0')}';
  }

  Book _stubBookForExport(BookNote note) {
    // M3.4 simplification: NotesPage fetches BookNote but not Book separately.
    // For Markdown frontmatter we synthesize a minimal Book. M3.5 will wire
    // real Book metadata.
    return Book(
      id: note.bookId,
      title: note.title.replaceFirst('《', '').replaceFirst('》读书笔记', ''),
      author: null,
      format: BookFormat.epub,
      fileSizeBytes: 0,
      addedAt: note.createdAt,
    );
  }
}
```

**Type consistency:** `PageBreak` constructor uses `startOffset:`/`endOffset:` named params everywhere. `MarkdownExporter.export` uses `book:` + `bookNote:` + `now:` everywhere. `AnnotationTile` uses `VoidCallback?` (not `ValueChanged<String>`) — matches existing.
