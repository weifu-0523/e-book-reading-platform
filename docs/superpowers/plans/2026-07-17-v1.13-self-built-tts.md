# v1.13 自研 Android TTS Plugin Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace `audio_service` + `audio_session` + `flutter_tts` on Android with a self-built MethodChannel plugin that lazy-initializes on first user gesture, restoring full TTS (basic read + notification + background play + lock screen + bluetooth) without triggering Chinese ROM startup kill.

**Architecture:** Custom FlutterPlugin (`ZhiYueTtsPlugin`) with empty `onAttachedToEngine` (the core invariant from v1.12.3-alpha lessons) + Service (`ZhiYueTtsService`) that owns TextToSpeech + MediaSession + AudioFocus. Dart side adds `SelfBuiltTtsServiceImpl implements TtsService` mirroring the existing `FlutterTtsServiceImpl` API. iOS interface slot preserved. Build flavor `withFgs`/`noFgs` provides fallback if manifest FGS declaration triggers ROM kill.

**Tech Stack:** Dart (Riverpod,现有 `TtsService` 接口) / Kotlin (Service, MediaSessionCompat, AudioFocusRequest, foreground service) / Flutter MethodChannel + EventChannel / Robolectric (Kotlin unit tests) / Gradle product flavors

**Spec:** `docs/superpowers/specs/2026-07-17-v1.13-self-built-tts-design.md`

import 'package:flutter/foundation.dart';

import 'flutter_tts_service.dart';
import 'self_built_tts_service.dart';
import 'tts_service.dart';

/// Creates a [TtsService] appropriate for [platform].
///
/// - Android: [SelfBuiltTtsServiceImpl] (v1.13 — avoids plugin attach-time
///   work that triggers Chinese ROM startup kill).
/// - iOS: [FlutterTtsServiceImpl] as fallback (v1.14 will add a native
///   AVSpeechSynthesizer impl mirroring the Android one).
/// - Windows/macOS/Linux: [FlutterTtsServiceImpl] (flutter_tts wraps SAPI /
///   NSSpeechSynthesizer / speech-dispatcher).
TtsService createTtsServiceForPlatform(String platform) {
  switch (platform.toLowerCase()) {
    case 'android':
      return SelfBuiltTtsServiceImpl();
    // TODO(v1.14): case 'ios': return IosAvSpeechTtsServiceImpl();
    default:
      return FlutterTtsServiceImpl();
  }
}

/// Production entry point — uses [defaultTargetPlatform].
TtsService createTtsService() {
  final target = defaultTargetPlatform;
  final platformName = switch (target) {
    TargetPlatform.android => 'android',
    TargetPlatform.iOS => 'ios',
    TargetPlatform.macOS => 'macos',
    TargetPlatform.windows => 'windows',
    TargetPlatform.linux => 'linux',
    TargetPlatform.fuchsia => 'fuchsia',
  };
  return createTtsServiceForPlatform(platformName);
}

`app/lib/services/tts/self_built_tts_service.dart`：

```dart
// SPDX-License-Identifier: AGPL-3.0-or-later
// Copyright (C) 2026 ZhiYue Open Source contributors

import 'dart:async';

import 'package:flutter/services.dart';

import 'tts_service.dart';

/// v1.13 Android TTS implementation via custom MethodChannel plugin.
///
/// **Critical invariant (v1.12.3-alpha 4-attempt lesson):** the Kotlin-side
/// `ZhiYueTtsPlugin.onAttachedToEngine` does NOTHING. All system bindings
/// (TextToSpeech, MediaSession, AudioFocus, startForeground) happen inside
/// `ZhiYueTtsService.onCreate`, which only runs after the user clicks
/// 「朗读」 (user-gesture exemption for Android 12+ FGS).
///
/// Dart side just routes calls through MethodChannel `dev.zhiyue.tts` and
/// subscribes to EventChannel `dev.zhiyue.tts/events` for state/progress.
class SelfBuiltTtsServiceImpl implements TtsService {
  static const MethodChannel _methodChannel =
      MethodChannel('dev.zhiyue.tts');
  static const EventChannel _eventChannel =
      EventChannel('dev.zhiyue.tts/events');

  final _stateController = StreamController<TtsState>.broadcast();
  final _progressController = StreamController<TtsProgress>.broadcast();
  final _errorController = StreamController<String>.broadcast();

  bool _initialized = false;
  int _currentSentenceIdx = 0;
  StreamSubscription? _eventSub;

  @override
  Stream<TtsState> get state => _stateController.stream;

  @override
  Stream<TtsProgress> get progress => _progressController.stream;

  @override
  Stream<String> get error => _errorController.stream;

  @override
  bool get isInitialized => _initialized;

  @override
  Future<void> init() async {
    if (_initialized) return;
    _eventSub = _eventChannel.receiveBroadcastStream().listen(_handleEvent);
    try {
      await _methodChannel.invokeMethod('init');
      _initialized = true;
      _stateController.add(TtsState.idle);
    } catch (e) {
      _errorController.add('init failed: $e');
      _stateController.add(TtsState.error);
      rethrow;
    }
  }

  void _handleEvent(dynamic event) {
    final map = event as Map<dynamic, dynamic>;
    final type = map['event'] as String?;
    switch (type) {
      case 'state':
        final stateStr = map['state'] as String?;
        _stateController.add(_parseState(stateStr));
        break;
      case 'progress':
        _progressController.add(TtsProgress(
          sentenceIndex: _currentSentenceIdx,
          charStart: map['start'] as int?,
          charEnd: map['end'] as int?,
        ));
        break;
      case 'error':
        _errorController.add(map['message'] as String? ?? 'unknown');
        _stateController.add(TtsState.error);
        break;
    }
  }

  TtsState _parseState(String? s) {
    switch (s) {
      case 'speaking':
        return TtsState.speaking;
      case 'paused':
        return TtsState.paused;
      case 'idle':
      case 'stopped':
        return TtsState.idle;
      case 'error':
        return TtsState.error;
      default:
        return TtsState.idle;
    }
  }

  @override
  Future<void> speakSentence(Sentence sentence) async {
    if (!_initialized) await init();
    await _methodChannel.invokeMethod('speak', {
      'text': sentence.text,
      'locale': sentence.locale,
      'sentenceIndex': _currentSentenceIdx,
    });
  }

  @override
  Future<void> stop() => _methodChannel.invokeMethod('stop');

  @override
  Future<void> pause() => _methodChannel.invokeMethod('pause');

  Future<void> resume() => _methodChannel.invokeMethod('resume');

  @override
  void setCurrentSentenceIndex(int idx) => _currentSentenceIdx = idx;

  @override
  Future<void> setSpeechRate(double rate) =>
      _methodChannel.invokeMethod('setRate', {'rate': rate});

  @override
  Future<void> setPitch(double pitch) =>
      _methodChannel.invokeMethod('setPitch', {'pitch': pitch});

  @override
  Future<void> setVolume(double volume) =>
      _methodChannel.invokeMethod('setVolume', {'volume': volume});

  @override
  Future<bool> isLanguageAvailable(String locale) async {
    final result = await _methodChannel.invokeMethod('isLanguageAvailable', {'locale': locale});
    return result == true;
  }

  @override
  Future<bool> isLanguageInstalled(String locale) async {
    final result = await _methodChannel.invokeMethod('isLanguageInstalled', {'locale': locale});
    return result == true;
  }

  @override
  Future<String?> getDefaultEngine() async {
    final result = await _methodChannel.invokeMethod('getDefaultEngine');
    return result is String ? result : null;
  }

  @override
  Future<void> dispose() async {
    await _eventSub?.cancel();
    await _stateController.close();
    await _progressController.close();
    await _errorController.close();
    _initialized = false;
  }
}
```

void main() {
  TestWidgetsFlutterBinding.ensureInitialized();

  late SelfBuiltTtsServiceImpl svc;
  final List<MethodCall> methodCalls = [];
  // Event sink captured for emitting fake events.
  EventSink<dynamic>? capturedSink;

  setUp(() {
    methodCalls.clear();
    capturedSink = null;
    svc = SelfBuiltTtsServiceImpl();

    MethodChannel('dev.zhiyue.tts').setMockMethodCallHandler((call) async {
      methodCalls.add(call);
      if (call.method == 'init') return null;
      if (call.method == 'isLanguageAvailable') return true;
      if (call.method == 'isLanguageInstalled') return true;
      if (call.method == 'getDefaultEngine') return 'com.google.android.tts';
      return null;
    });

    EventChannel('dev.zhiyue.tts/events')
        .setMockStreamHandlerFactory((_) => _MockStreamHandler(() => capturedSink));
  });

  tearDown(() async {
    await svc.dispose();
    MethodChannel('dev.zhiyue.tts').setMockMethodCallHandler(null);
    EventChannel('dev.zhiyue.tts/events').setMockStreamHandlerFactory(null);
  });

  test('init invokes MethodChannel init', () async {
    await svc.init();
    expect(methodCalls, contains(predicate<MethodCall>((c) => c.method == 'init')));
    expect(svc.isInitialized, isTrue);
  });

  test('init failure propagates error and sets state to error', () async {
    MethodChannel('dev.zhiyue.tts').setMockMethodCallHandler((call) async {
      throw PlatformException(code: 'TTS_UNAVAILABLE', message: 'no engine');
    });
    final errors = <String>[];
    svc.error.listen(errors.add);
    await expectLater(svc.init(), throwsA(isA<PlatformException>()));
    expect(errors, isNotEmpty);
  });

  test('speakSentence routes through MethodChannel with text+locale', () async {
    await svc.init();
    await svc.speakSentence(const Sentence(
      text: '你好',
      locale: 'zh-CN',
      charStart: 0,
      charEnd: 2,
    ));
    expect(
      methodCalls,
      contains(predicate<MethodCall>((c) =>
          c.method == 'speak' &&
          (c.arguments as Map)['text'] == '你好' &&
          (c.arguments as Map)['locale'] == 'zh-CN')),
    );
  });

  test('EventChannel state events mapped to TtsState', () async {
    final states = <TtsState>[];
    svc.state.listen(states.add);
    await svc.init();
    capturedSink?.add({'event': 'state', 'state': 'speaking'});
    capturedSink?.add({'event': 'state', 'state': 'paused'});
    capturedSink?.add({'event': 'state', 'state': 'idle'});
    await Future<void>.delayed(Duration.zero);
    expect(states, contains(TtsState.speaking));
    expect(states, contains(TtsState.paused));
    expect(states, contains(TtsState.idle));
  });

  test('EventChannel progress events mapped to TtsProgress', () async {
    final progress = <TtsProgress>[];
    svc.progress.listen(progress.add);
    await svc.init();
    svc.setCurrentSentenceIndex(3);
    capturedSink?.add({'event': 'progress', 'start': 0, 'end': 5});
    await Future<void>.delayed(Duration.zero);
    expect(progress, isNotEmpty);
    expect(progress.last.sentenceIndex, 3);
    expect(progress.last.charStart, 0);
    expect(progress.last.charEnd, 5);
  });

  test('setSpeechRate / setPitch / setVolume routed', () async {
    await svc.init();
    await svc.setSpeechRate(0.5);
    await svc.setPitch(1.2);
    await svc.setVolume(0.8);
    expect(methodCalls.any((c) => c.method == 'setRate' && (c.arguments as Map)['rate'] == 0.5), isTrue);
    expect(methodCalls.any((c) => c.method == 'setPitch' && (c.arguments as Map)['pitch'] == 1.2), isTrue);
    expect(methodCalls.any((c) => c.method == 'setVolume' && (c.arguments as Map)['volume'] == 0.8), isTrue);
  });

class _MockStreamHandler implements EventSink<dynamic> {

  @override

import android.media.AudioAttributes
import android.media.AudioFocusRequest
import android.media.AudioManager
import android.os.Build
import android.telephony.PhoneStateListener
import android.telephony.TelephonyCallback
import android.telephony.TelephonyManager
import android.content.Context

class ZhiYueTtsService : Service() {
    private var audioFocusRequest: AudioFocusRequest? = null
    private var audioManager: AudioManager? = null
    private var phoneStateListener: PhoneStateListener? = null
    private var telephonyCallback: TelephonyCallback? = null

    internal var audioFocusRequestedForTest = false
        private set
    internal var phoneStateListenerRegisteredForTest = false
        private set

    override fun onCreate() {
        super.onCreate()
        // ... existing TextToSpeech + MediaSession setup ...
        setupAudioFocus()
        setupPhoneStateListener()
    }

    private fun setupAudioFocus() {
        audioManager = getSystemService(Context.AUDIO_SERVICE) as AudioManager
        val attrs = AudioAttributes.Builder()
            .setUsage(AudioAttributes.USAGE_MEDIA)
            .setContentType(AudioAttributes.CONTENT_TYPE_SPEECH)
            .build()
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            audioFocusRequest = AudioFocusRequest.Builder(AudioManager.AUDIOFOCUS_GAIN)
                .setAudioAttributes(attrs)
                .setOnAudioFocusChangeListener { focusChange ->
                    when (focusChange) {
                        AudioManager.AUDIOFOCUS_LOSS,
                        AudioManager.AUDIOFOCUS_LOSS_TRANSIENT -> handlePause()
                        // No auto-resume on focus gain (let user re-click)
                    }
                }
                .build()
            audioManager?.requestAudioFocus(audioFocusRequest!!)
        } else {
            @Suppress("DEPRECATION")
            audioManager?.requestAudioFocus(
                null,
                AudioManager.STREAM_MUSIC,
                AudioManager.AUDIOFOCUS_GAIN
            )
        }
        audioFocusRequestedForTest = true
    }

    private fun setupPhoneStateListener() {
        val tm = getSystemService(Context.TELEPHONY_SERVICE) as TelephonyManager
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S) {
            telephonyCallback = object : TelephonyCallback(), TelephonyCallback.CallStateListener {
                override fun onCallStateChanged(state: Int) {
                    if (state == TelephonyManager.CALL_STATE_RINGING) handlePause()
                }
            }
            tm.registerTelephonyCallback(mainLooper, telephonyCallback!!)
        } else {
            phoneStateListener = object : PhoneStateListener() {
                @Deprecated("deprecated in API 31")
                override fun onCallStateChanged(state: Int, phoneNumber: String?) {
                    if (state == TelephonyManager.CALL_STATE_RINGING) handlePause()
                }
            }
            tm.listen(phoneStateListener, PhoneStateListener.LISTEN_CALL_STATE)
        }
        phoneStateListenerRegisteredForTest = true
    }

    override fun onDestroy() {
        // ... existing tts + mediaSession cleanup ...
        audioFocusRequest?.let { audioManager?.abandonAudioFocusRequest(it) }
        audioFocusRequest = null
        audioFocusRequestedForTest = false

        val tm = getSystemService(Context.TELEPHONY_SERVICE) as TelephonyManager
        telephonyCallback?.let {
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S) tm.unregisterTelephonyCallback(it)
        }
        phoneStateListener?.let { tm.listen(it, PhoneStateListener.LISTEN_NONE) }
        telephonyCallback = null
        phoneStateListener = null
        phoneStateListenerRegisteredForTest = false

        super.onDestroy()
    }
}

import android.app.Notification
import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.PendingIntent
import android.content.pm.ServiceInfo
import android.os.Build
import androidx.core.app.NotificationCompat

class ZhiYueTtsService : Service() {
    private var isForeground = false

    internal val isForegroundForTest: Boolean
        get() = isForeground

    override fun onCreate() {
        super.onCreate()
        // ... existing TextToSpeech + MediaSession + AudioFocus setup ...

        // LAST: start foreground (must complete within 5s for Android 12+ ANR)
        ensureNotificationChannel()
        startForeground(NOTIFICATION_ID, buildNotification("智阅朗读", "准备朗读"),
            ServiceInfo.FOREGROUND_SERVICE_TYPE_MEDIA_PLAYBACK)
        isForeground = true
    }

    private fun ensureNotificationChannel() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(
                CHANNEL_ID,
                "智阅朗读",
                NotificationManager.IMPORTANCE_LOW
            ).apply {
                description = "朗读控制通知"
                setShowBadge(false)
            }
            val nm = getSystemService(NOTIFICATION_SERVICE) as NotificationManager
            nm.createNotificationChannel(channel)
        }
    }

    private fun buildNotification(title: String, content: String): Notification {
        val pauseIntent = PendingIntent.getService(
            this, 0,
            Intent(this, ZhiYueTtsService::class.java).setAction(ACTION_PAUSE),
            PendingIntent.FLAG_IMMUTABLE
        )
        val stopIntent = PendingIntent.getService(
            this, 1,
            Intent(this, ZhiYueTtsService::class.java).setAction(ACTION_STOP),
            PendingIntent.FLAG_IMMUTABLE
        )
        return NotificationCompat.Builder(this, CHANNEL_ID)
            .setSmallIcon(android.R.drawable.ic_media_play)
            .setContentTitle(title)
            .setContentText(content)
            .setOngoing(true)
            .addAction(android.R.drawable.ic_media_pause, "暂停", pauseIntent)
            .addAction(android.R.drawable.ic_menu_close_clear_cancel, "停止", stopIntent)
            .setStyle(androidx.media.app.NotificationCompat.MediaStyle()
                .setMediaSession(mediaSession?.sessionId))
            .build()
    }

    private fun handleStop() {
        tts?.stop()
        updatePlaybackState(PlaybackStateCompat.STATE_STOPPED)
        mediaSession?.isActive = false
        if (isForeground) {
            stopForeground(true)
            isForeground = false
        }
    }

    override fun onDestroy() {
        if (isForeground) {
            stopForeground(true)
            isForeground = false
        }
        // ... existing cleanup ...
        super.onDestroy()
    }

    companion object {
        private const val NOTIFICATION_ID = 0x7T_00 // 32000+; unique per service
        private const val CHANNEL_ID = "zhiyue_tts"
        // ... existing ACTION_* constants ...
    }

import android.content.Intent
import android.os.Build

class ZhiYueTtsPlugin : FlutterPlugin, MethodChannel.MethodCallHandler {
    internal var contextForTest: Context? = null
        private set

    fun attachContextForTest(ctx: Context) {
        contextForTest = ctx
    }

    override fun onMethodCall(call: MethodCall, result: MethodChannel.Result) {
        val ctx = contextForTest ?: run {
            result.error("NO_CONTEXT", "Plugin not attached", null)
            return
        }
        when (call.method) {
            "init" -> {
                ensureServiceStarted(ctx)
                result.success(null)
            }
            "speak" -> {
                ensureServiceStarted(ctx)
                val args = call.arguments as Map<String, Any>
                val intent = Intent(ctx, ZhiYueTtsService::class.java)
                    .setAction(ZhiYueTtsService.ACTION_SPEAK)
                    .putExtra(ZhiYueTtsService.EXTRA_TEXT, args["text"] as String)
                    .putExtra(ZhiYueTtsService.EXTRA_LOCALE, args["locale"] as String)
                    .putExtra(ZhiYueTtsService.EXTRA_SENTENCE_INDEX,
                        (args["sentenceIndex"] as Number).toInt())
                ctx.startService(intent)
                result.success(null)
            }
            "pause" -> forwardAction(ctx, ZhiYueTtsService.ACTION_PAUSE, result)
            "resume" -> forwardAction(ctx, ZhiYueTtsService.ACTION_RESUME, result)
            "stop" -> forwardAction(ctx, ZhiYueTtsService.ACTION_STOP, result)
            "setRate" -> { /* T13 — set on service via intent extra or shared prefs */ result.success(null) }
            "setPitch" -> { result.success(null) }
            "setVolume" -> { result.success(null) }
            "setLanguage" -> { result.success(null) }
            "isLanguageAvailable" -> result.success(true)
            "isLanguageInstalled" -> result.success(true)
            "getDefaultEngine" -> result.success("com.google.android.tts")
            else -> result.notImplemented()
        }
    }

    private fun ensureServiceStarted(ctx: Context) {
        if (!isServiceCreated) {
            val intent = Intent(ctx, ZhiYueTtsService::class.java)
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
                ctx.startForegroundService(intent)
            } else {
                ctx.startService(intent)
            }
            isServiceCreated = true
        }
    }

    private fun forwardAction(ctx: Context, action: String, result: MethodChannel.Result) {
        ctx.startService(Intent(ctx, ZhiYueTtsService::class.java).setAction(action))
        result.success(null)
    }
}
