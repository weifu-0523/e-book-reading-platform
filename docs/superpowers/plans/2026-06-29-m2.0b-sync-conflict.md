```dart
import 'package:drift/drift.dart';

/// One row per sync conflict detected during pull.
///
/// A "conflict" means: remote record's mtime was NOT strictly after the
/// local row's mtime, so the applier skipped the replacement. We log it
/// so users can see which writes lost (LWW is silent otherwise).
///
/// Purge policy: rows older than 30 days are removed by SyncController
/// after each successful syncNow().
class SyncConflicts extends Table {
  IntColumn get id => integer().autoIncrement()();
  TextColumn get recordId => text()();
  TextColumn get recordType => text()(); // 'annotation' | 'bookNote' | 'readingProgress'
  DateTimeColumn get localMtime => dateTime()();
  DateTimeColumn get remoteMtime => dateTime()();
  TextColumn get winnerDevice => text()(); // always local deviceId (LWW semantics)
  TextColumn get loserDevice => text()();  // remote snapshot source deviceId
  DateTimeColumn get detectedAt => dateTime()();
  TextColumn get payloadPreview => text()(); // remote payload JSON, truncated to 200 chars
}
```

```dart
// app/lib/data/local/daos/sync_conflicts_dao.dart
import 'package:drift/drift.dart';

import '../database.dart';
import '../tables/sync_conflicts.dart';

part 'sync_conflicts_dao.g.dart';

@DriftAccessor(tables: [SyncConflicts])
class SyncConflictsDao extends DatabaseAccessor<AppDatabase>
    with _$SyncConflictsDaoMixin {
  SyncConflictsDao(super.db);

  Future<int> insert({
    required String recordId,
    required String recordType,
    required DateTime localMtime,
    required DateTime remoteMtime,
    required String winnerDevice,
    required String loserDevice,
    required DateTime detectedAt,
    required String payloadPreview,
  }) {
    return into(syncConflicts).insert(
      SyncConflictsCompanion.insert(
        recordId: recordId,
        recordType: recordType,
        localMtime: localMtime,
        remoteMtime: remoteMtime,
        winnerDevice: winnerDevice,
        loserDevice: loserDevice,
        detectedAt: detectedAt,
        payloadPreview: payloadPreview,
      ),
    );
  }

  /// Most recent N conflicts, newest first.
  Future<List<SyncConflictsData>> recent(int limit) {
    return (select(syncConflicts)
          ..orderBy([(t) => OrderingTerm.desc(t.detectedAt)])
          ..limit(limit))
        .get();
  }

  /// Delete rows with detected_at strictly before cutoff. Returns count removed.
  Future<int> purgeBefore(DateTime cutoff) {
    return (delete(syncConflicts)
          ..where((t) => t.detectedAt.isSmallerThanValue(cutoff)))
        .go();
  }

  Future<int> clear() {
    return delete(syncConflicts).go();
  }

  Future<int> count() async {
    final row = await customSelect(
      'SELECT COUNT(*) AS c FROM sync_conflicts',
    ).getSingle();
    return row.read<int>('c');
  }
}
```

```dart
// app/lib/services/sync/conflict_sink.dart
import 'package:zhiyue/services/sync/sync_record.dart';

/// Conflict events emitted by RemoteRecordApplier when LWW drops a remote
/// record (local mtime is newer or equal). Implementations decide where to
/// persist (DB table, test list, dev stderr, etc.).
///
/// Why abstract: future v2 could swap in a Stream emitter without touching
/// the applier.
abstract class ConflictSink {
  Future<void> record({
    required String recordId,
    required SyncRecordType recordType,
    required DateTime localMtime,
    required DateTime remoteMtime,
    required String remoteDeviceId,
    required Map<String, dynamic> remotePayload,
  });
}

/// No-op sink. Default for M2.0a backward-compat (existing applier tests
/// construct without a sink).
class NullConflictSink implements ConflictSink {
  const NullConflictSink();

  @override
  Future<void> record({
    required String recordId,
    required SyncRecordType recordType,
    required DateTime localMtime,
    required DateTime remoteMtime,
    required String remoteDeviceId,
    required Map<String, dynamic> remotePayload,
  }) async {}
}

/// In-memory list sink for tests. Exposed publicly so other test files
/// (applier, engine) can import it.
class ListConflictSink implements ConflictSink {
  final List<ConflictRecord> records = [];

  @override
  Future<void> record({
    required String recordId,
    required SyncRecordType recordType,
    required DateTime localMtime,
    required DateTime remoteMtime,
    required String remoteDeviceId,
    required Map<String, dynamic> remotePayload,
  }) async {
    records.add(ConflictRecord(
      recordId: recordId,
      recordType: recordType,
      localMtime: localMtime,
      remoteMtime: remoteMtime,
      remoteDeviceId: remoteDeviceId,
      remotePayload: remotePayload,
    ));
  }
}

/// Plain data class captured by ListConflictSink.
class ConflictRecord {
  const ConflictRecord({
    required this.recordId,
    required this.recordType,
    required this.localMtime,
    required this.remoteMtime,
    required this.remoteDeviceId,
    required this.remotePayload,
  });

  final String recordId;
  final SyncRecordType recordType;
  final DateTime localMtime;
  final DateTime remoteMtime;
  final String remoteDeviceId;
  final Map<String, dynamic> remotePayload;
}
```

```dart
/// Applies remote [SyncRecord]s to the local DB with last-write-wins
/// semantics. Idempotent — safe to call on records already applied.
///
/// When a remote record is skipped because local is newer-or-equal, the
/// applier emits a conflict event via [conflictSink] (default
/// [NullConflictSink] = silent). Books are excluded (no updated_at column).
class RemoteRecordApplier {
  RemoteRecordApplier(this._db, {ConflictSink? conflictSink})
      : _conflictSink = conflictSink ?? const NullConflictSink();

  final AppDatabase _db;
  final ConflictSink _conflictSink;

  /// Set by WebDavSyncEngine per snapshot file before applying its lines.
  /// Falls back to 'unknown' if never set (defensive).
  String currentRemoteDeviceId = 'unknown';

  Future<bool> apply(SyncRecord record) async {
    switch (record.type) {
      case SyncRecordType.annotation:
        return _applyAnnotation(record);
      case SyncRecordType.bookNote:
        return _applyBookNote(record);
      case SyncRecordType.readingProgress:
        return _applyReadingProgress(record);
      case SyncRecordType.book:
        return _applyBook(record);
    }
  }

  Future<bool> _applyAnnotation(SyncRecord record) async {
    final existing = await _db.customSelect(
      'SELECT updated_at FROM annotations WHERE id = ?',
      variables: [Variable.withString(record.id)],
    ).getSingleOrNull();
    if (existing != null) {
      final localMtime = existing.read<DateTime>('updated_at');
      if (!record.mtime.isAfter(localMtime)) {
        await _recordConflict(record, localMtime);
        return false;
      }
    }

    if (record.deleted) {
      await _db.customStatement(
        'DELETE FROM annotations WHERE id = ?',
        [record.id],
      );
      return true;
    }

    final p = record.payload;
    final bookNoteId = p['bookNoteId'] as String;
    final locator = p['locator'] as String;
    final quote = p['quote'] as String;
    final note = p['note'] as String?;
    final color = p['color'] as String;
    final tags = p['tags'] as String;
    final source = p['source'] as String? ?? 'manual';

    await _db.customInsert(

      variables: [
        Variable.withString(record.id),
        Variable.withString(bookNoteId),
        Variable.withString(locator),
        Variable.withString(quote),
        Variable<String>(note),
        Variable.withString(color),
        Variable.withString(tags),
        Variable.withString(source),
        Variable.withDateTime(record.mtime),
        Variable.withDateTime(record.mtime),
      ],
    );
    return true;
  }

  Future<bool> _applyBookNote(SyncRecord record) async {
    final existing = await _db.customSelect(
      'SELECT updated_at FROM book_notes WHERE id = ?',
      variables: [Variable.withString(record.id)],
    ).getSingleOrNull();

      return true;

        remotePayload: record.payload,
      );
    } catch (_) {
      // Silent — diagnostic sink failure must not fail sync.
    }
  }
}
```

```dart
      return const SyncResult.success(pushedCount: 0, pulledCount: 0);

      existing = await _rateLimiter.run(() => webdav.getString(_snapshotPath));
    } on WebDavException catch (e) {
      if (e.statusCode != 404) rethrow;
      // 404 = file doesn't exist yet, treat as empty

    if (existing.isNotEmpty) {
      lines.addAll(existing.split('\n').where((l) => l.trim().isNotEmpty));
    }
    for (final row in pending) {
      lines.add(row.payloadJson);
    }
    final newBody = '${lines.join('\n')}\n';

    await _rateLimiter.run(() => webdav.putString(_snapshotPath, newBody));

    final now = DateTime.now();
    for (final row in pending) {
      await outbox.markPushed(row.id, now);
    }

    return SyncResult.success(pushedCount: pending.length, pulledCount: 0);
  }

    final applier = RemoteRecordApplier(database, conflictSink: _conflictSink);
    final List<String> files;
    try {
      files = await _rateLimiter.run(
          () => webdav.listFiles('/zhiyue/snapshots/'));
    } on WebDavException catch (e) {
      if (e.statusCode == 404) {
        // Directory doesn't exist yet — no remote data.
        return const SyncResult.success(pushedCount: 0, pulledCount: 0);
      }
      rethrow;
    }

    var applied = 0;
    for (final file in files) {
      if (file == '$deviceId.jsonl') continue; // skip own file
      final fileDeviceId = file.replaceAll('.jsonl', '');
      applier.currentRemoteDeviceId = fileDeviceId; // per-file context
      final body = await _rateLimiter.run(
          () => webdav.getString('/zhiyue/snapshots/$file'));
      final lines = body.split('\n').where((l) => l.trim().isNotEmpty);

          final json = jsonDecode(line) as Map<String, dynamic>;
          final record = SyncRecord.fromJson(json);
          final didApply = await applier.apply(record);
          if (didApply) applied++;
        } catch (_) {
          // Skip malformed line — don't fail entire sync.
        }
      }
    }
    return SyncResult.success(pushedCount: 0, pulledCount: applied);

      pushedCount: pushResult.pushedCount,
```

```dart
// app/lib/services/sync/background_sync_scheduler.dart
import 'dart:async';

/// Periodic background sync driver with lifecycle + cooldown awareness.
///
/// - Timer.periodic with [interval] (default 5 minutes).
/// - [pause] cancels the active timer (app backgrounded).
/// - [resume] triggers an immediate tick then restarts the periodic timer.
/// - skipIfRunning guard prevents overlap with manual syncNow.
/// - After onTick throws, scheduler enters a 5-minute cooldown; manual
///   syncNow is NOT subject to this cooldown.
class BackgroundSyncScheduler {
  BackgroundSyncScheduler({
    required this.onTick,
    required DateTime Function() clock,
    Duration initialInterval = const Duration(minutes: 5),
  })  : _clock = clock,
        _interval = initialInterval;

  final Future<void> Function() onTick;
  final DateTime Function() _clock;
  Duration _interval;

  Timer? _timer;
  bool _isRunning = false;
  DateTime? _lastErrorAt;

  static const _cooldown = Duration(minutes: 5);

  bool get isRunning => _isRunning;

  /// Start periodic ticking. Safe to call multiple times — second call is
  /// a no-op if a timer is already active.
  void start() {
    _resetTimer();
  }

  /// Stop periodic ticking (e.g. app backgrounded).
  void pause() {
    _timer?.cancel();
    _timer = null;
  }

  /// Trigger one immediate tick, then restart the periodic timer.
  /// Used when app returns to foreground.
  Future<void> resume() async {
    await _tickSafe();
    _resetTimer();
  }

  /// Change the periodic interval. Duration.zero stops the timer.
  void updateInterval(Duration interval) {
    _interval = interval;
    if (interval == Duration.zero) {
      pause();
    } else {
      _resetTimer();
    }
  }

  void _resetTimer() {
    _timer?.cancel();
    if (_interval == Duration.zero) return;
    _timer = Timer.periodic(_interval, (_) => _tickSafe());
  }

  Future<void> _tickSafe() async {
    if (_isRunning) return;
    if (_lastErrorAt != null &&
        _clock().difference(_lastErrorAt!) < _cooldown) {
      return;
    }
    _isRunning = true;
    try {
      await onTick();
      _lastErrorAt = null;
    } catch (_) {
      _lastErrorAt = _clock();
      // Swallow — manual syncNow surfaces errors via state; scheduler
      // runs in the background without UI listeners.
    } finally {
      _isRunning = false;
    }
  }
}
```

```dart
/// Thin wrapper around connectivity_plus for testability.
///
/// Strategy: connectivity_plus is used as a "fast offline hint" — if it
/// reports none, we skip the WebDAV round-trip. If it reports wifi/ethernet
/// but the network is actually down, the existing WebDavClient try/catch
/// still handles it. So this wrapper is non-blocking.
abstract class ConnectivityService {
  Future<bool> get isOnline;
}

class ConnectivityPlusService implements ConnectivityService {
  ConnectivityPlusService([Connectivity? connectivity])
      : _connectivity = connectivity ?? Connectivity();

  final Connectivity _connectivity;

  @override
  Future<bool> get isOnline async {
    try {
      final result = await _connectivity.checkConnectivity();
      return result != ConnectivityResult.none;
    } catch (_) {
      // connectivity_plus can throw on some Windows virtual NIC setups.
      // Fail-open: assume online; let WebDavClient try/catch catch real errors.
      return true;
    }
  }
}
```

```dart
/// Persisted background-sync interval (Duration).
///
/// Stored under `zhiyue.sync.intervalMinutes` as integer minutes.
/// Duration.zero means "off" (manual sync only). Default on first load
/// is 5 minutes.
class SyncIntervalPref {

      : _storage = storage ?? const FlutterSecureStorage();

  static const _key = 'zhiyue.sync.intervalMinutes';
  static const _defaultMinutes = 5;

  final FlutterSecureStorage _storage;

  Future<Duration> load() async {
    final raw = await _storage.read(key: _key);
    if (raw == null) return const Duration(minutes: _defaultMinutes);
    final minutes = int.tryParse(raw);
    if (minutes == null || minutes < 0) {
      return const Duration(minutes: _defaultMinutes);
    }
    return Duration(minutes: minutes);
  }

  Future<void> save(Duration interval) async {
    final minutes = interval.inMinutes;
    await _storage.write(key: _key, value: minutes.toString());
  }

  Future<void> clear() async {
    await _storage.delete(key: _key);
  }
}
```

```dart
        status: SyncStatus.error,
        lastMessage: 'syncCredentialsMissing',

      return;

          now - const Duration(days: 30));

        lastSyncAt: now,
        lastMessage: result.error,
        lastPushedCount: result.pushedCount,
        lastPulledCount: result.pulledCount,
        conflictCount: state.conflictCount,
        currentInterval: state.currentInterval,
      );

        status: SyncStatus.error,
        lastSyncAt: _clock(),
        lastMessage:
            e.statusCode == 429 ? 'syncRateLimitedWarning' : e.toString(),
      );
    } catch (e) {
      state = state.copyWith(
        status: SyncStatus.error,
        lastSyncAt: _clock(),
        lastMessage: e.toString(),
      );
    }
  }

  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (state == AppLifecycleState.resumed) {
      _scheduler.resume();
    } else if (state == AppLifecycleState.paused) {
      _scheduler.pause();
    }
  }

    WidgetsBinding.instance.removeObserver(this);
    _scheduler.pause();
    super.dispose();
  }
```
