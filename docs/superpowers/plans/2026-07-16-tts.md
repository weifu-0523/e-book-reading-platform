```dart
import 'dart:async';

enum TtsState { idle, loading, speaking, paused, error }

class TtsProgress {
  final int sentenceIndex;
  final int? charStart;
  final int? charEnd;
  const TtsProgress({required this.sentenceIndex, this.charStart, this.charEnd});
}

class Sentence {
  final String text;
  final String locale;
  final int charStart;
  final int charEnd;
  final String? startCfi;
  final String? endCfi;

  const Sentence({
    required this.text,
    required this.locale,
    required this.charStart,
    required this.charEnd,
    this.startCfi,
    this.endCfi,
  });
}

abstract class TtsService {
  Stream<TtsState> get state;
  Stream<TtsProgress> get progress;
  Stream<String> get error;

  bool get isInitialized;

  Future<void> init();
  Future<void> speakSentence(Sentence sentence);
  Future<void> stop();
  Future<void> pause();

  Future<void> setSpeechRate(double rate);  // -1.0 ~ 1.0
  Future<void> setPitch(double pitch);      // 0.5 ~ 2.0
  Future<void> setVolume(double volume);    // 0.0 ~ 1.0

  Future<bool> isLanguageAvailable(String locale);
  Future<bool> isLanguageInstalled(String locale);

  Future<void> dispose();
}
```

```dart
// SPDX-License-Identifier: AGPL-3.0-or-later
// Copyright (C) 2026 ZhiYue Open Source contributors

import 'dart:async';

import 'package:flutter_tts/flutter_tts.dart';

import 'tts_service.dart';

class FlutterTtsServiceImpl implements TtsService {
  final FlutterTts _tts = FlutterTts();
  final _stateController = StreamController<TtsState>.broadcast();
  final _progressController = StreamController<TtsProgress>.broadcast();
  final _errorController = StreamController<String>.broadcast();

  bool _initialized = false;
  int _currentSentenceIdx = 0;

  @override
  Stream<TtsState> get state => _stateController.stream;

  @override
  Stream<TtsProgress> get progress => _progressController.stream;

  @override
  Stream<String> get error => _errorController.stream;

  @override
  bool get isInitialized => _initialized;

  @override
  Future<void> init() async {
    if (_initialized) return;
    _tts.setInitHandler(() => _stateController.add(TtsState.idle));
    _tts.setStartHandler(() => _stateController.add(TtsState.speaking));
    _tts.setCompletionHandler(() {
      _stateController.add(TtsState.idle);
    });
    _tts.setCancelHandler(() => _stateController.add(TtsState.idle));
    _tts.setProgressHandler((text, start, end, word) {
      _progressController.add(TtsProgress(
        sentenceIndex: _currentSentenceIdx,
        charStart: start,
        charEnd: end,
      ));
    });
    _tts.setErrorHandler((msg) {
      _errorController.add(msg);
      _stateController.add(TtsState.error);
    });
    await _tts.setAwaitSpeakCompletion(true);
    _stateController.add(TtsState.idle);
    _initialized = true;
  }

  @override
  Future<void> speakSentence(Sentence sentence) async {
    if (!_initialized) await init();
    await _tts.setLanguage(sentence.locale);
    await _tts.speak(sentence.text);
  }

  /// Called by TtsPlaybackController to update sentence index for progress reporting.
  void setCurrentSentenceIndex(int idx) => _currentSentenceIdx = idx;

  @override
  Future<void> stop() => _tts.stop();

  @override
  Future<void> pause() => _tts.pause();

  @override
  Future<void> setSpeechRate(double rate) => _tts.setSpeechRate(rate);

  @override
  Future<void> setPitch(double pitch) => _tts.setPitch(pitch);

  @override
  Future<void> setVolume(double volume) => _tts.setVolume(volume);

  @override
  Future<bool> isLanguageAvailable(String locale) =>
      _tts.isLanguageAvailable(locale);

  @override
  Future<bool> isLanguageInstalled(String locale) async {
    final installed = await _tts.isLanguageInstalled(locale);
    return installed == true;
  }

  @override
  Future<void> dispose() async {
    await _tts.stop();
    await _stateController.close();
    await _progressController.close();
    await _errorController.close();
    _initialized = false;
  }
}
```

```dart
// SPDX-License-Identifier: AGPL-3.0-or-later
// Copyright (C) 2026 ZhiYue Open Source contributors

/// Unicode 启发式 locale 检测。
///
/// 规则：CJK Unified Ideographs (0x4E00-0x9FFF) + CJK Extension A
/// (0x3400-0x4DBF) 占比 > 30% → zh-CN，否则 en-US。
class LocaleDetector {
  const LocaleDetector._();

  static String detect(String text) {
    if (text.isEmpty) return 'en-US';
    final cjkCount = text.runes.where((c) =>
        (c >= 0x4E00 && c <= 0x9FFF) || (c >= 0x3400 && c <= 0x4DBF)).length;
    if (cjkCount > text.length * 0.3) return 'zh-CN';
    return 'en-US';
  }
}
```

```dart
// SPDX-License-Identifier: AGPL-3.0-or-later
// Copyright (C) 2026 ZhiYue Open Source contributors

import 'tts_service.dart';
import 'locale_detector.dart';

/// CFI 范围（start/end 是章节纯文本的 char offset）。
typedef ParagraphCfiRange = ({String startCfi, String endCfi});
typedef ParagraphCharSpan = ({int start, int end});

/// 句子切分器：按中英标点 + 段落边界切。
class SentenceSegmenter {
  const SentenceSegmenter._();

  static const int _maxSentenceLength = 200;

  static const _punctuation = RegExp(r'[。！？\.!?;；]');

  static List<Sentence> split(
    String text, {
    List<ParagraphCfiRange>? paragraphCfiRanges,
    List<ParagraphCharSpan>? paragraphCharSpans,
  }) {
    if (text.isEmpty) return const [];

    final List<Sentence> result = [];
    int cursor = 0;

    while (cursor < text.length) {
      // 找下一个段落边界
      final nextParaNewline = text.indexOf('\n', cursor);
      final paraEnd = nextParaNewline == -1 ? text.length : nextParaNewline;

      // 在当前段落内找句号
      var matchEnd = _findSentenceEnd(text, cursor, paraEnd);

      // 如果找到的句子超过 max，强制切
      while (matchEnd - cursor > _maxSentenceLength) {
        final forceEnd = cursor + _maxSentenceLength;
        _emit(result, text, cursor, forceEnd, paragraphCfiRanges, paragraphCharSpans);
        cursor = forceEnd;
      }

      _emit(result, text, cursor, matchEnd, paragraphCfiRanges, paragraphCharSpans);
      cursor = matchEnd;

      // 跳过空白和换行
      while (cursor < text.length &&
          (text[cursor] == '\n' ||
              text[cursor] == ' ' ||
              text[cursor] == '\t' ||
              text[cursor] == '\r')) {
        cursor++;
      }
    }

    return result;
  }

  static int _findSentenceEnd(String text, int start, int paraEnd) {
    // 在 [start, paraEnd) 范围内找标点
    for (var i = start; i < paraEnd; i++) {
      if (_punctuation.hasMatch(text[i])) {
        // 包含标点本身和后续空格
        var end = i + 1;
        while (end < paraEnd &&
            (text[end] == ' ' || text[end] == '\t')) {
          end++;
        }
        return end;
      }
    }
    return paraEnd;
  }

  static void _emit(
    List<Sentence> result,
    String text,
    int start,
    int end,
    List<ParagraphCfiRange>? cfiRanges,
    List<ParagraphCharSpan>? charSpans,
  ) {
    if (start >= end) return;
    final sentenceText = text.substring(start, end);
    final locale = LocaleDetector.detect(sentenceText);

    String? startCfi;
    String? endCfi;
    if (cfiRanges != null && charSpans != null) {
      for (var i = 0; i < charSpans.length; i++) {
        final span = charSpans[i];
        if (start >= span.start && start < span.end) {
          startCfi = cfiRanges[i].startCfi;
        }
        if (end > span.start && end <= span.end) {
          endCfi = cfiRanges[i].endCfi;
          break;
        }
      }
    }

    result.add(Sentence(
      text: sentenceText,
      locale: locale,
      charStart: start,
      charEnd: end,
      startCfi: startCfi,
      endCfi: endCfi,
    ));
  }
}
```

```dart
import 'dart:async';

import 'package:flutter_riverpod/flutter_riverpod.dart';

import '../../../services/tts/tts_service.dart';
import '../../../core/providers.dart' as core;

enum TtsPhase { idle, loading, speaking, paused, error }

sealed class TtsScope {
  const TtsScope();
}
class TtsScopeAiMessage extends TtsScope {
  final String messageId;
  const TtsScopeAiMessage({required this.messageId});
}
class TtsScopeChapter extends TtsScope {
  final String bookId;
  final int chapterIdx;
  const TtsScopeChapter({required this.bookId, required this.chapterIdx});
}

class TtsPlaybackState {
  final TtsPhase phase;
  final TtsScope? scope;
  final int currentIndex;
  final int queueLength;
  final String? errorMessage;

  const TtsPlaybackState({
    this.phase = TtsPhase.idle,
    this.scope,
    this.currentIndex = 0,
    this.queueLength = 0,
    this.errorMessage,
  });

  TtsPlaybackState copyWith({
    TtsPhase? phase,
    TtsScope? scope,
    int? currentIndex,
    int? queueLength,
    String? errorMessage,
    bool clearScope = false,
  }) {
    return TtsPlaybackState(
      phase: phase ?? this.phase,
      scope: clearScope ? null : (scope ?? this.scope),
      currentIndex: currentIndex ?? this.currentIndex,
      queueLength: queueLength ?? this.queueLength,
      errorMessage: errorMessage,
    );
  }
}

final ttsServiceProvider = Provider<TtsService>((ref) {
  final svc = core.globalTtsServiceOverride ?? FlutterTtsServiceImpl();
  ref.onDispose(svc.dispose);
  return svc;
});

class TtsPlaybackController extends StateNotifier<TtsPlaybackState> {
  final TtsService _service;
  List<Sentence> _queue = [];
  StreamSubscription? _stateSub;

  TtsPlaybackController(this._service) : super(const TtsPlaybackState()) {
    _stateSub = _service.state.listen(_onServiceState);
  }

  Future<void> play({
    required TtsScope scope,
    required List<Sentence> sentences,
  }) async {
    if (sentences.isEmpty) return;
    await _service.init();
    _queue = List.unmodifiable(sentences);
    state = TtsPlaybackState(
      phase: TtsPhase.loading,
      scope: scope,
      currentIndex: 0,
      queueLength: _queue.length,
    );
    await _speakCurrent();
  }

  Future<void> pause() async {
    await _service.pause();
    state = state.copyWith(phase: TtsPhase.paused);
  }

  Future<void> resume() async {
    state = state.copyWith(phase: TtsPhase.loading);
    await _speakCurrent();
  }

  Future<void> stop() async {
    await _service.stop();
    _queue = [];
    state = const TtsPlaybackState();
  }

  Future<void> skipNext() async {
    if (state.currentIndex + 1 < _queue.length) {
      state = state.copyWith(currentIndex: state.currentIndex + 1);
      await _service.stop();
      await _speakCurrent();
    } else {
      await stop();
    }
  }

  Future<void> skipPrevious() async {
    if (state.currentIndex > 0) {
      state = state.copyWith(currentIndex: state.currentIndex - 1);
      await _service.stop();
      await _speakCurrent();
    }
  }

  Future<void> _speakCurrent() async {
    if (_queue.isEmpty) return;
    final sentence = _queue[state.currentIndex];
    try {
      await _service.speakSentence(sentence);
    } catch (e) {
      state = state.copyWith(phase: TtsPhase.error, errorMessage: e.toString());
    }
  }

  void _onServiceState(TtsState svcState) {
    if (svcState == TtsState.idle && state.phase == TtsPhase.speaking) {
      // current sentence completed → advance
      if (state.currentIndex + 1 < _queue.length) {
        state = state.copyWith(
            currentIndex: state.currentIndex + 1, phase: TtsPhase.loading);
        _speakCurrent();
      } else {
        // queue exhausted
        _queue = [];
        state = const TtsPlaybackState();
      }
    } else if (svcState == TtsState.speaking &&
        state.phase == TtsPhase.loading) {
      state = state.copyWith(phase: TtsPhase.speaking);
    }
  }

  @override
  void dispose() {
    _stateSub?.cancel();
    super.dispose();
  }
}

final ttsPlaybackControllerProvider =
    StateNotifierProvider<TtsPlaybackController, TtsPlaybackState>(
  (ref) => TtsPlaybackController(ref.watch(ttsServiceProvider)),
);
```

```dart
class TtsSettingsNotifier extends StateNotifier<TtsSettings> {
  final Directory _dir;
  File? _file;

  TtsSettingsNotifier(this._dir) : super(const TtsSettings()) {
    _load();
  }

  File get _settingsFile {
    _file ??= File('${_dir.path}/tts_settings.json');
    return _file!;
  }

  Future<void> _load() async {
    try {
      if (await _settingsFile.exists()) {
        final raw = await _settingsFile.readAsString();
        final j = jsonDecode(raw) as Map<String, dynamic>;
        state = TtsSettings.fromJson(j);
      }
    } catch (_) {
      // corrupted file — keep defaults
    }
  }

  Future<void> _persist() async {
    try {
      await _settingsFile.writeAsString(jsonEncode(state.toJson()));
    } catch (_) {
      // best-effort
    }
  }

  Future<void> setRate(double v) async {
    state = state.copyWith(rate: v);
    await _persist();
  }

  Future<void> setPitch(double v) async {
    state = state.copyWith(pitch: v);
    await _persist();
  }

  Future<void> setVolume(double v) async {
    state = state.copyWith(volume: v);
    await _persist();
  }

  Future<void> setChapterBoundary(TtsChapterBoundary mode) async {
    state = state.copyWith(chapterBoundary: mode);
    await _persist();
  }
}
```

```dart
  Widget build(BuildContext context, WidgetRef ref) {

          ),

              } else {
                notifier.pause();
```

```dart
class TtsReaderController extends StateNotifier<TtsReaderState> {
  TtsReaderController(this._ref) : super(...);

  Future<void> startCurrentChapter() async {
    final book = _ref.read(currentBookProvider);
    final chapterIdx = _ref.read(currentChapterIdxProvider);
    final text = await _ref.read(chapterTextProvider(book.id, chapterIdx).future);
    final cfiRanges = await CFIBoundaryExtractor.extract(...);
    final sentences = SentenceSegmenter.split(
      text,
      paragraphCfiRanges: cfiRanges,
      paragraphCharSpans: ...,
    );

    // 订阅 ttsPlaybackController 的 currentIndex 变化
    _ref.listen(ttsPlaybackControllerProvider, (prev, next) {
      _onPlaybackChange(next, sentences);
    });

    await _ref.read(ttsPlaybackControllerProvider.notifier).play(
      scope: TtsScopeChapter(bookId: book.id, chapterIdx: chapterIdx),
      sentences: sentences,
    );
  }

  void _onPlaybackChange(TtsPlaybackState playback, List<Sentence> sentences) {
    if (playback.phase != TtsPhase.speaking) return;
    final sentence = sentences[playback.currentIndex];

    // 1. auto-flip page if needed
    final targetPage = _readerEngine.locatorToPage(sentence.startCfi);
    if (targetPage != _currentPage) {
      _readerEngine.goToPage(targetPage);
    }

    // 2. show highlight overlay
    _ref.read(temporaryHighlightProvider.notifier).show(
      sentence.startCfi!, sentence.endCfi!,
    );

    // 3. update last_tts_position
    _ref.read(booksDaoProvider).setLastTtsPosition(bookId, LastTtsPosition(
      chapterIdx: chapterIdx,
      sentenceIdx: playback.currentIndex,
      updatedAt: DateTime.now().millisecondsSinceEpoch,
    ));
  }
}
```

```dart
  final TtsService _tts;

      controls: [MediaControl.skipToPrevious, MediaControl.pause, MediaControl.skipToNext, MediaControl.stop],
      playing: true,
      processingState: AudioProcessingState.ready,

  @override
```

```dart
enum TtsAvailability { ready, engineMissing, voiceDownloadRequired, samsungRestricted }

Future<TtsAvailability> checkTtsAvailability() async {
  final tts = FlutterTts();
  final defaultEngine = await tts.getDefaultEngine();
  if (SamsungTtsDetector.isSamsungEngine(defaultEngine)) {
    return TtsAvailability.samsungRestricted;
  }
  final zhAvailable = await tts.isLanguageAvailable('zh-CN');
  final enAvailable = await tts.isLanguageAvailable('en-US');
  if (zhAvailable != true && enAvailable != true) {
    return TtsAvailability.engineMissing;
  }
  final zhInstalled = await tts.isLanguageInstalled('zh-CN');
  if (zhInstalled != true && zhAvailable == true) {
    return TtsAvailability.voiceDownloadRequired;
  }
  return TtsAvailability.ready;
}

class SamsungTtsDetector {
  static bool isSamsungEngine(String? engineName) {
    return engineName == 'com.samsung.SMT';
  }
}
```
