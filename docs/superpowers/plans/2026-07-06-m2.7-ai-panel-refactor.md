// SPDX-License-Identifier: AGPL-3.0-or-later
import 'package:drift/drift.dart';
import '../database.dart';
import '../tables/chat_sessions.dart';

part 'chat_sessions_dao.g.dart';

@DriftAccessor(tables: [ChatSessions])
class ChatSessionsDao extends DatabaseAccessor<AppDatabase>
    with _$ChatSessionsDaoMixin {
  ChatSessionsDao(super.db);

  Future<List<ChatSession>> listForBook(String bookId) {
    final q = select(chatSessions)
      ..where((t) => t.bookId.equals(bookId) & t.deletedAt.isNull())
      ..orderBy([(t) => OrderingTerm.desc(t.updatedAt)]);
    return q.get();
  }

  Future<ChatSession?> findById(String id) {
    return (select(chatSessions)..where((t) => t.id.equals(id)))
        .getSingleOrNull();
  }

  Future<String> insertSession({
    required String id,
    required String bookId,
    required String title,
    String mode = 'normal',
    required DateTime createdAt,
  }) async {
    await into(chatSessions).insert(ChatSessionsCompanion.insert(
      id: id,
      bookId: bookId,
      title: title,
      mode: Value(mode),
      createdAt: createdAt,
      updatedAt: createdAt,
    ));
    return id;
  }

  Future<void> touch(String id, DateTime at) {
    return (update(chatSessions)..where((t) => t.id.equals(id)))
        .write(ChatSessionsCompanion(updatedAt: Value(at)));
  }

  Future<void> setMode(String id, String mode, DateTime at) {
    return (update(chatSessions)..where((t) => t.id.equals(id)))
        .write(ChatSessionsCompanion(mode: Value(mode), updatedAt: Value(at)));
  }

  Future<void> rename(String id, String title, DateTime at) {
    return (update(chatSessions)..where((t) => t.id.equals(id)))
        .write(ChatSessionsCompanion(title: Value(title), updatedAt: Value(at)));
  }

  Future<void> softDelete(String id, DateTime at) {
    return (update(chatSessions)..where((t) => t.id.equals(id)))
        .write(ChatSessionsCompanion(deletedAt: Value(at), updatedAt: Value(at)));
  }
}

import 'package:drift/drift.dart';
import '../database.dart';
import '../tables/chat_messages.dart';

part 'chat_messages_dao.g.dart';

@DriftAccessor(tables: [ChatMessages])
class ChatMessagesDao extends DatabaseAccessor<AppDatabase>
    with _$ChatMessagesDaoMixin {
  ChatMessagesDao(super.db);

  Future<List<ChatMessage>> listForSession(String sessionId) {
    final q = select(chatMessages)
      ..where((t) => t.sessionId.equals(sessionId) & t.deletedAt.isNull())
      ..orderBy([(t) => OrderingTerm.asc(t.createdAt)]);
    return q.get();
  }

  Stream<List<ChatMessage>> watchForSession(String sessionId) {
    final q = select(chatMessages)
      ..where((t) => t.sessionId.equals(sessionId) & t.deletedAt.isNull())
      ..orderBy([(t) => OrderingTerm.asc(t.createdAt)]);
    return q.watch();
  }

  Future<String> insertMessage({
    required String id,
    required String sessionId,
    required String role,
    required String content,
    String? presetSource,
    String? attachmentNoteId,
    String? attachmentNoteSnapshot,
    required DateTime createdAt,
  }) async {
    await into(chatMessages).insert(ChatMessagesCompanion.insert(
      id: id,
      sessionId: sessionId,
      role: role,
      content: content,
      presetSource: Value(presetSource),
      attachmentNoteId: Value(attachmentNoteId),
      attachmentNoteSnapshot: Value(attachmentNoteSnapshot),
      createdAt: createdAt,
      updatedAt: createdAt,
    ));
    return id;
  }

  Future<void> appendContent(String id, String delta, DateTime at) {
    // 流式 token 追加用 raw SQL（drift 不直接支持 append）
    return customStatement(
      'UPDATE chat_messages SET content = content || ?, updated_at = ? WHERE id = ?',
      [Variable(delta), Variable(at.toInt()), Variable(id)],
    );
  }

  Future<void> softDelete(String id, DateTime at) {
    return (update(chatMessages)..where((t) => t.id.equals(id)))
        .write(ChatMessagesCompanion(deletedAt: Value(at), updatedAt: Value(at)));
  }

  Future<void> deleteAllForSession(String sessionId) {
    return (delete(chatMessages)
          ..where((t) => t.sessionId.equals(sessionId)))
        .go();
  }
}

```dart
// SPDX-License-Identifier: AGPL-3.0-or-later
import 'package:drift/drift.dart';
import '../database.dart';
import 'chat_messages_dao.dart';
import 'chat_sessions_dao.dart';

/// UUID 由调用方传入（方便测试），Repository 不关心生成策略。
class ChatRepository {
  ChatRepository(this._db);
  final AppDatabase _db;

  ChatSessionsDao get _sessions => _db.attach(_db.chatSessionsDao);
  ChatMessagesDao get _messages => _db.attach(_db.chatMessagesDao);

  Future<String> createSession({
    required String bookId,
    required String title,
    String mode = 'normal',
    String? id,
    DateTime? now,
  }) async {
    final at = now ?? DateTime.now().toUtc();
    final sid = id ?? _uuid();
    await _sessions.insertSession(
      id: sid,
      bookId: bookId,
      title: title,
      mode: mode,
      createdAt: at,
    );
    return sid;
  }

  Future<ChatSession?> getSession(String id) => _sessions.findById(id);
  Future<List<ChatSession>> listSessionsForBook(String bookId) =>
      _sessions.listForBook(bookId);
  Future<List<ChatMessage>> listMessages(String sessionId) =>
      _messages.listForSession(sessionId);
  Stream<List<ChatMessage>> watchMessages(String sessionId) =>
      _messages.watchForSession(sessionId);

  Future<String> appendMessage({
    required String sessionId,
    required String role,
    required String content,
    String? presetSource,
    String? attachmentNoteId,
    String? attachmentNoteSnapshot,
    DateTime? now,
  }) async {
    final at = now ?? DateTime.now().toUtc();
    final mid = _uuid();
    await _messages.insertMessage(
      id: mid,
      sessionId: sessionId,
      role: role,
      content: content,
      presetSource: presetSource,
      attachmentNoteId: attachmentNoteId,
      attachmentNoteSnapshot: attachmentNoteSnapshot,
      createdAt: at,
    );
    await _sessions.touch(sessionId, at);
    return mid;
  }

  Future<void> appendContentToMessage(String id, String delta) {
    return _messages.appendContent(id, delta, DateTime.now().toUtc());
  }

  Future<void> setSessionMode(String sessionId, String mode) {
    return _sessions.setMode(sessionId, mode, DateTime.now().toUtc());
  }

  Future<void> renameSession(String sessionId, String title) {
    return _sessions.rename(sessionId, title, DateTime.now().toUtc());
  }

  Future<void> clearSession(String sessionId) async {
    await _messages.deleteAllForSession(sessionId);
    await _sessions.touch(sessionId, DateTime.now().toUtc());
  }

  Future<void> softDeleteSession(String sessionId) async {
    await _messages.deleteAllForSession(sessionId);
    await _sessions.softDelete(sessionId, DateTime.now().toUtc());
  }

  String _uuid() {
    // 简单时间戳 + 随机；生产可换 uuid package
    return '${DateTime.now().microsecondsSinceEpoch}-${_rand.nextInt(1 << 32)}';
  }
  static final _rand = _Random();

  // 持有引用以避免每次访问都创建
  ChatSessionsDao get chatSessionsDao => _sessions;
  ChatMessagesDao get chatMessagesDao => _messages;

// SPDX-License-Identifier: AGPL-3.0-or-later

class ChatTurn {
  const ChatTurn({required this.role, required this.content});
  final String role; // 'user' | 'ai' | 'system'
  final String content;
}

class PromptBuilderExt {
  static const _socraticSystemPrompt = '''你是苏格拉底式导师。
规则：
- 不要直接给出答案
- 通过反问、追问引导学生自己思考
- 检测到学生疲惫或想结束时，做一段本章收束回顾
- 始终用中文''';

  static const _normalSystemPrompt = '''你是一个阅读助手，回答用户关于本书的问题。''';

  /// 自由聊天 prompt：system + 全量历史 + 当前问题。
  /// 完整记忆模式（B），用户用 /new /clear 自己控制大小。
  static String buildChatPrompt({
    required List<ChatTurn> history,
    required String mode,
    String? pendingUserContent,
    String? attachmentSnapshot,
  }) {
    final sys = mode == 'socratic' ? _socraticSystemPrompt : _normalSystemPrompt;
    final buf = StringBuffer('$sys\n\n');
    for (final turn in history) {
      if (turn.role == 'system') continue; // 跳过旧的 system markers
      final speaker = turn.role == 'user' ? '用户' : 'AI';
      buf.writeln('$speaker: ${turn.content}');
    }
    if (attachmentSnapshot != null && attachmentSnapshot.isNotEmpty) {
      buf.writeln('\n[用户附加笔记]');
      buf.writeln(attachmentSnapshot);
    }
    if (pendingUserContent != null && pendingUserContent.isNotEmpty) {
      buf.writeln('\n用户: $pendingUserContent');
      buf.writeln('AI:'); // 引导模型续写
    }
    return buf.toString();
  }

  /// 苏格拉底 bootstrap：recap + 要求提一个开放性问题
  static String buildSocraticBootstrapPrompt({required String recap}) {
    return '''$_socraticSystemPrompt

下面是本章回顾：
$recap

请基于其中任意一个关键点，向学生提一个开放性问题（不要直接讲解）。
用中文提问。''';
  }

  /// 章节回顾 prompt：复用 BookReviewer (M2.1) 的输出风格。
  /// 这里只封装给 RunRecapUseCase 用。
  static String buildRecapPrompt({required String chapterText}) {
    return '''请回顾以下章节内容，列出 3-5 个关键点（带简短说明）：

$chapterText

输出格式：
• 关键点 1 - 说明
• 关键点 2 - 说明
...''';
  }
}

// SPDX-License-Identifier: AGPL-3.0-or-later
import 'dart:async';
import '../../../data/local/daos/chat_repository.dart';
import '../../../services/ai/prompt_builder_ext.dart';

/// 抽象 LLM 客户端：use case 不直接依赖 OpenAI client，便于测试
abstract class ChatLlmClient {
  Stream<String> streamChat(String prompt);
  Future<void> cancel();
}

/// 用户自由消息：写 user msg → 取历史 → build prompt → stream → 边收边写 AI msg
class SendMessageUseCase {
  SendMessageUseCase({required this.repo, required this.llm});
  final ChatRepository repo;
  final ChatLlmClient llm;

  /// 流式发射增量 chunk；完成时 last 是完整文本。
  Stream<String> run({
    required String sessionId,
    required String userContent,
    String? attachmentNoteId,
    String? attachmentNoteSnapshot,
  }) async* {
    // 1. 写 user message
    await repo.appendMessage(
      sessionId: sessionId,
      role: 'user',
      content: userContent,
      attachmentNoteId: attachmentNoteId,
      attachmentNoteSnapshot: attachmentNoteSnapshot,
    );

    // 2. 取历史 + 当前 session mode
    final history = await repo.listMessages(sessionId);
    final session = await repo.getSession(sessionId);
    final mode = session?.mode ?? 'normal';

    // 3. build prompt（去掉刚写的 user msg，避免重复——builder 会重新拼）
    final turns = history
        .where((m) => m.id != history.last.id) // 去掉刚 append 的
        .map((m) => ChatTurn(role: m.role, content: m.content))
        .toList();

    final prompt = PromptBuilderExt.buildChatPrompt(
      history: turns,
      mode: mode,
      pendingUserContent: userContent,
      attachmentSnapshot: attachmentSnapshot,
    );

    // 4. 写空的 AI message 占位，边收边 appendContent
    final aiMsgId = await repo.appendMessage(
      sessionId: sessionId,
      role: 'ai',
      content: '',
    );

    final buf = StringBuffer();
    await for (final chunk in llm.streamChat(prompt)) {
      buf.write(chunk);
      await repo.appendContentToMessage(aiMsgId, chunk);
      yield chunk;
    }
  }
}

import 'dart:async';
import 'dart:convert';
import 'package:crypto/crypto.dart';
import '../../../data/local/daos/ai_summaries_dao.dart';
import '../../../data/local/daos/chat_repository.dart';

/// Subagent pattern: 不读 ChatSessionController 状态，不读 chat_messages 历史。
/// 只接受 (bookId, chapterIndex) → 跑 LLM → 写一条 AI message 进当前 session。
class RunChapterGuideUseCase {
  RunChapterGuideUseCase({
    required this.repo,
    required this.llm,
    required this.cache,
    required this.chapterTextProvider,
  });
  final ChatRepository repo;
  final ChatLlmClient llm;
  final AiSummaryCache cache; // 已有，从 M1.5a
  final Future<String> Function(String bookId, int chapterIndex) chapterTextProvider;

  /// [source] = 'card.chapter' 或 '/chapter'，仅作为持久化的标记
  Stream<String> run({
    required String sessionId,
    required String bookId,
    required int chapterIndex,
    required String source,
  }) async* {
    final chapterText = await chapterTextProvider(bookId, chapterIndex);
    final sourceHash = sha256.convert(utf8.encode(chapterText)).toString();

    // 缓存命中：直接 emit cached + 写 message
    final cached = await cache.get(bookId: bookId, mode: 'chapter', sourceHash: sourceHash);
    if (cached != null) {
      await repo.appendMessage(
        sessionId: sessionId,
        role: 'ai',
        content: cached,
        presetSource: source,
      );
      yield cached;

class RunRecapUseCase {
  RunRecapUseCase({
    required this.repo,
    required this.cache,
    required this.chapterTextProvider,
    required this.reviewer,
  });
  final ChatRepository repo;
  final AiSummaryCache cache;
  final Future<String> Function(String bookId, int chapterIndex) chapterTextProvider;
  final BookReviewer reviewer; // M2.1 既有

  Stream<String> run({
    required String sessionId,
    required String bookId,
    required int chapterIndex,
  }) async* {
    final chapterText = await chapterTextProvider(bookId, chapterIndex);
    final sourceHash = sha256.convert(utf8.encode(chapterText)).toString();
    final cached = await cache.get(bookId: bookId, mode: 'recap', sourceHash: sourceHash);
    final aiMsgId = await repo.appendMessage(
      sessionId: sessionId,
      role: 'ai',
      content: '',
      presetSource: '/recap',
    );

    if (cached != null) {
      await repo.appendContentToMessage(aiMsgId, cached);
      yield cached;
      return;
    }

    final buf = StringBuffer();
    await for (final chunk in reviewer.runRecap(bookId: bookId, chapterIndex: chapterIndex)) {
      buf.write(chunk);
      await repo.appendContentToMessage(aiMsgId, chunk);
      yield chunk;
    }
    await cache.put(
      bookId: bookId, mode: 'recap', sourceHash: sourceHash,
      summaryText: buf.toString(), modelName: '', baseUrl: '',
    );
  }
}

class SessionLifecycleUseCases {
  SessionLifecycleUseCases({required this.repo});

    return repo.createSession(bookId: bookId, title: title, mode: mode);

  /// 删除当前 session，返回应该激活的下一个 session id（最近 updated），
  /// 没有可用 session 时返回 null。
  Future<String?> delete(String sessionId) async {
    final session = await repo.getSession(sessionId);
    if (session == null) return null;
    final bookId = session.bookId;
    await repo.softDeleteSession(sessionId);
    final remaining = await repo.listSessionsForBook(bookId);
    if (remaining.isEmpty) return null;
    remaining.sort((a, b) => b.updatedAt.compareTo(a.updatedAt));
    return remaining.first.id;
  }
}

  }) {

class ChatSessionController extends FamilyNotifier<ChatSessionState, String> {
  late String _bookId;
  StreamSubscription<List<ChatMessage>>? _msgSub;

  @override
  ChatSessionState build(String bookId) {
    _bookId = bookId;
    ref.onDispose(() => _msgSub?.cancel());
    return const ChatSessionState();
  }

  Future<void> selectSession(String sessionId) async {
    state = state.copyWith(currentSessionId: sessionId, error: null);
    _msgSub?.cancel();
    _msgSub = ref.read(chatRepositoryProvider).watchMessages(sessionId).listen((messages) {
      state = state.copyWith(messages: messages);
    });
  }

  Future<void> ensureSession() async {
    if (state.currentSessionId != null) return;
    final sessions = await ref.read(chatRepositoryProvider).listSessionsForBook(_bookId);
    if (sessions.isEmpty) {
      final id = await ref.read(sessionLifecycleUseCaseProvider).create(bookId: _bookId, title: _defaultTitle());
      await selectSession(id);
    } else {
      await selectSession(sessions.first.id);
    }
  }

  Future<void> sendMessage({required String text, String? attachmentNoteId, String? attachmentSnapshot}) async {
    final sid = state.currentSessionId;
    if (sid == null) return;
    state = state.copyWith(isStreaming: true, error: null);
    try {
      await ref.read(sendMessageUseCaseProvider).run(
        sessionId: sid,
        userContent: text,
        attachmentNoteId: attachmentNoteId,
        attachmentNoteSnapshot: attachmentSnapshot,
      ).last;
      state = state.copyWith(isStreaming: false);
    } catch (e) {
      state = state.copyWith(isStreaming: false, error: e.toString());
    }
  }

  Future<void> runChapterGuide({required int chapterIndex, required String source}) async {
    final sid = state.currentSessionId;
    if (sid == null) return;
    state = state.copyWith(isStreaming: true, error: null);
    try {
      await ref.read(runChapterGuideUseCaseProvider).run(
        sessionId: sid, bookId: _bookId, chapterIndex: chapterIndex, source: source,
      ).last;
      state = state.copyWith(isStreaming: false);
    } catch (e) {
      state = state.copyWith(isStreaming: false, error: e.toString());
    }
  }

  Future<void> runSocraticBootstrap({required int chapterIndex}) async {
    final sid = state.currentSessionId;
    if (sid == null) return;
    state = state.copyWith(isStreaming: true, error: null);
    try {
      await ref.read(switchSessionModeUseCaseProvider).toSocratic(sessionId: sid);
      await ref.read(runSocraticBootstrapUseCaseProvider).run(
        sessionId: sid, bookId: _bookId, chapterIndex: chapterIndex, source: 'card.socratic',
      ).last;
      state = state.copyWith(isStreaming: false);
    } catch (e) {
      state = state.copyWith(isStreaming: false, error: e.toString());
    }
  }

  Future<void> runRecap({required int chapterIndex}) async {
    final sid = state.currentSessionId;
    if (sid == null) return;
    state = state.copyWith(isStreaming: true, error: null);
    try {
      await ref.read(runRecapUseCaseProvider).run(
        sessionId: sid, bookId: _bookId, chapterIndex: chapterIndex,
      ).last;
      state = state.copyWith(isStreaming: false);
    } catch (e) {
      state = state.copyWith(isStreaming: false, error: e.toString());
    }
  }

  Future<void> exitSocratic() async {
    final sid = state.currentSessionId;
    if (sid == null) return;
    await ref.read(switchSessionModeUseCaseProvider).toNormal(sessionId: sid);
  }

  String _defaultTitle() {
    final now = DateTime.now();
    return '新会话 ${now.year}-${now.month.toString().padLeft(2, '0')}-${now.day.toString().padLeft(2, '0')}';
  }

  final text = _controller.text;
  final newtext = text.substring(0, text.length - 1); // 移除末尾 @
  _controller.text = newtext;
  _controller.selection = TextSelection.fromPosition(TextPosition(offset: newtext.length));
  setState(() {
    if (_pendingAttachment != null) {
      ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('已替换附件'), duration: Duration(seconds: 1)));
    }
    _pendingAttachment = selected;
    _showNotePopup = false;
  });
}
```
