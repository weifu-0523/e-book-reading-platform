# M2.8 跨端同步范围扩展 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让用户在电脑上看的书和聊天可以在手机上接着，反之亦然。同步范围从 6 种扩到 9 种 + 书 binary 文件。

**Architecture:** 沿用现有 per-device JSONL snapshot + last-write-wins。Phase 2 metadata-only 同步扩 SyncRecordType enum + 3 个 applier。Phase 3 binary 文件走独立 WebDAV 路径 `/zhiyue/files/<bookId>.<ext>`，与 metadata 并行。

**Tech Stack:** Flutter 3.44 / drift 2.28 / Riverpod 2.5 / http 1.2 / WebDAV

**Spec:** [docs/superpowers/specs/2026-07-12-m28-sync-scope-design.md](../specs/2026-07-12-m28-sync-scope-design.md)

```dart
factory SyncRecord.fromJson(Map<String, dynamic> json) {
  final typeName = json['type'] as String;
  final type = SyncRecordType.values.asNameMap()[typeName];
  if (type == null) {
    throw FormatException('Unknown SyncRecordType: $typeName');
  }
  return SyncRecord(
    id: json['id'] as String,
    type: type,
    mtime: DateTime.parse(json['mtime'] as String),
    deleted: json['deleted'] as bool,
    payload: Map<String, dynamic>.from(json['payload'] as Map),
  );
}
```

```dart
Future<void> recordBookTag({
  required String bookId,
  required String tag,
}) {
  final compositeId = '${bookId}__${tag}';
  final payload = <String, dynamic>{
    'bookId': bookId,
    'tag': tag,
  };
  return _enqueue(SyncRecord(
    id: compositeId,
    type: SyncRecordType.bookTag,
    mtime: DateTime.now(),
    deleted: false,
    payload: payload,
  ));
}

Future<void> recordBookTagDeletion({
  required String bookId,
  required String tag,
}) {
  final compositeId = '${bookId}__${tag}';
  return _enqueue(SyncRecord.tombstone(
    id: compositeId,
    type: SyncRecordType.bookTag,
    mtime: DateTime.now(),
  ));
}
```

```dart
abstract class BookTagsRepository {
  Future<List<String>> tagsOfBook(String bookId);
  Future<Set<String>> allTags();
  Future<List<String>> bookIdsWithTag(String tag);
  Future<void> addTag({required String bookId, required String tag});
  Future<void> removeTag({required String bookId, required String tag});
  Future<void> replaceTagsForBook({required String bookId, required List<String> tags});
}

class LocalBookTagsRepository implements BookTagsRepository {
  LocalBookTagsRepository(this._db, {SyncRecorder? syncRecorder})
      : _tagsDao = BookTagsDao(_db),
        _syncRecorder = syncRecorder;
  final AppDatabase _db;
  final BookTagsDao _tagsDao;
  final SyncRecorder? _syncRecorder;

  @override
  Future<void> addTag({required String bookId, required String tag}) async {
    await _tagsDao.addTag(bookId: bookId, tag: tag);
    await _syncRecorder?.recordBookTag(bookId: bookId, tag: tag);
  }

  @override
  Future<void> removeTag({required String bookId, required String tag}) async {
    await _tagsDao.removeTag(bookId: bookId, tag: tag);
    await _syncRecorder?.recordBookTagDeletion(bookId: bookId, tag: tag);
  }

  @override
  Future<void> replaceTagsForBook({required String bookId, required List<String> tags}) async {
    final existing = await _tagsDao.tagsOfBook(bookId);
    final toAdd = tags.toSet().difference(existing.toSet());
    final toRemove = existing.toSet().difference(tags.toSet());
    await _tagsDao.replaceTagsForBook(bookId: bookId, tags: tags);
    for (final t in toAdd) {
      await _syncRecorder?.recordBookTag(bookId: bookId, tag: t);
    }
    for (final t in toRemove) {
      await _syncRecorder?.recordBookTagDeletion(bookId: bookId, tag: t);
    }
  }

  // read methods just delegate
  @override
  Future<List<String>> tagsOfBook(String bookId) => _tagsDao.tagsOfBook(bookId);
  @override
  Future<Set<String>> allTags() => _tagsDao.allTags();
  @override
  Future<List<String>> bookIdsWithTag(String tag) => _tagsDao.bookIdsWithTag(tag);
}
```

```dart
class BookFileSyncService {
  BookFileSyncService({
    required AppDatabase database,
    required WebDavClient webdav,
    required Directory booksDir,
  });

  static const _needsDownloadMarker = '__needs_download__';
  static const _filesDir = '/zhiyue/files';

  Future<void> uploadBookFile(String bookId) async {
    final book = await _fetchBook(bookId);
    if (book == null || book.fileSyncedAt != null) return;
    final file = File(book.filePath);
    if (!file.existsSync()) return; // can't upload what we don't have

    final bytes = await file.readAsBytes();
    final remotePath = '$_filesDir/$bookId.${book.format}';

    // Ensure remote dir exists (idempotent)
    await webdav.ensureAncestors('$remotePath/dummy');

    try {
      await webdav.putBytes(remotePath, bytes);
    } on WebDavException catch (e) {
      if (e.statusCode == 507) {
        // Quota exceeded — mark this book as quota-failed (separate column? or just rethrow)
        rethrow;
      }
      rethrow;
    }

    await _db.customStatement(
      'UPDATE books SET file_synced_at = ? WHERE id = ?',
      [DateTime.now().millisecondsSinceEpoch, bookId],
    );
  }

  Future<void> downloadBookFile(String bookId) async {
    final book = await _fetchBook(bookId);
    if (book == null) return;

    final remotePath = '$_filesDir/$bookId.${book.format}';
    final bytes = await webdav.getBytes(remotePath);

    final localPath = '${_booksDir.path}/$bookId.${book.format}';
    final file = File(localPath);
    await file.create(recursive: true);
    await file.writeAsBytes(bytes);

    await _db.customStatement(
      'UPDATE books SET file_path = ? WHERE id = ?',
      [localPath, bookId],
    );
  }

  Future<void> uploadPendingFiles() async {
    final rows = await _db.customSelect(
      'SELECT id FROM books WHERE file_synced_at IS NULL AND file_path != ?',
      variables: [Variable.withString(_needsDownloadMarker)],
    ).get();
    for (final row in rows) {
      final id = row.read<String>('id');
      try {
        await uploadBookFile(id);
      } catch (_) {
        // continue; one bad book shouldn't break the rest
      }
    }
  }

  Future<void> deleteRemoteFile(String bookId, String format) async {
    final remotePath = '$_filesDir/$bookId.$format';
    try {
      await webdav.delete(remotePath);
    } on WebDavException catch (e) {
      if (e.statusCode != 404) rethrow;
    }
  }

  Future<bool> localFileExists(String bookId) async {
    final book = await _fetchBook(bookId);
    if (book == null) return false;
    return File(book.filePath).existsSync();
  }
}
```
