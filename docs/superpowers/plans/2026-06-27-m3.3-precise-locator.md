Create `reader_engine/lib/rendering/html_text_utils.dart`:

```dart
import 'package:html/parser.dart';

/// Pure utility for converting HTML strings to plain text.
///
/// Used by [ReaderEngine.currentChapterPlainText] to derive a
/// deterministic character stream that matches what the user sees
/// rendered, enabling character-offset-based locators.
class HtmlTextUtils {
  /// Convert [html] to plain text by:
  /// 1. Parsing via `package:html`
  /// 2. Removing `<script>` and `<style>` subtrees
  /// 3. Walking all text nodes in document order
  /// 4. Concatenating their decoded text
  ///
  /// Returns empty string for null/empty input.
  static String htmlToPlainText(String html) {
    if (html.isEmpty) return '';
    final doc = HtmlParser(html).parse();
    // Remove script/style before walking
    doc.querySelectorAll('script, style').forEach((e) => e.remove());
    final buf = StringBuffer();
    _walkText(doc, buf);
    return buf.toString();
  }

  static void _walkText(dynamic /*Node*/ node, StringBuffer buf) {
    final nodes = node.nodes;
    for (final n in nodes) {
      if (n.nodeType == /*Node.TEXT_NODE*/ 3) {
        buf.write(n.text);
      } else if (n.nodeType == /*Node.ELEMENT_NODE*/ 1) {
        _walkText(n, buf);
      }
    }
  }
}
```

> **Note for implementer:** The `node.nodeType` integer constants (3 = TEXT_NODE, 1 = ELEMENT_NODE) are DOM spec values. If `package:html` exposes named constants, prefer those (e.g., `Node.TEXT_NODE`). Verify against `package:html` API during impl; if named constants exist use them.

  return plain;

import 'package:reader_engine/core/locator.dart';
import 'package:reader_engine/navigation/toc.dart';

/// Flattens a [TableOfContents] into a `{chapterIndex: title}` map.
///
/// Walk algorithm: depth-first traversal. For each [TocEntry]:
/// - If `entry.locator is EpubCfiLocator`, register
///   `loc.chapterIndex → entry.title`.
/// - Skip non-EpubCfiLocator entries (e.g., EpubRangeLocator).
/// - Duplicate chapterIndex: depth-first first-wins (matches reader order).
class TocTitleResolver {
  static Map<int, String> resolve(TableOfContents? toc) {
    if (toc == null) return {};
    final map = <int, String>{};
    void walk(List<TocEntry> entries) {
      for (final e in entries) {
        final loc = e.locator;
        if (loc is EpubCfiLocator) {
          map.putIfAbsent(loc.chapterIndex, () => e.title);
        }
        walk(e.children);
      }
    }
    walk(toc.entries);
    return map;
  }
}
```

/// Resolve user-selected plain text to an [EpubRangeLocator] carrying
/// real character offsets within the chapter plain text.
///
/// Algorithm: indexOf the selected text within [chapterPlainText].
/// Returns null if selectedText is empty or not found.
///
/// Multi-match: indexOf returns first hit (M3.3 MVP disambiguation).
/// M3.4 may upgrade to anchor-based disambiguation.
Future<EpubRangeLocator?> resolveSelectionLocator({
  required String selectedText,
  required int chapterIndex,
  required String chapterPlainText,
  String? chapterTitle,
}) async {
  if (selectedText.isEmpty) return null;
  final startOffset = chapterPlainText.indexOf(selectedText);
  if (startOffset < 0) return null;
  return EpubRangeLocator(
    chapterIndex: chapterIndex,
    chapterTitle: chapterTitle,
    startOffset: startOffset,
    endOffset: startOffset + selectedText.length,
    quote: selectedText,
  );
}
```

```dart
import 'package:flutter/material.dart';
import 'package:flutter/rendering.dart';

/// Scrolls a [Scrollable] to make the character at [targetOffset] visible.
///
/// Algorithm:
/// 1. Collect all [RenderParagraph] descendants of [scrollable] in DFS order.
/// 2. Accumulate each paragraph's `text.toPlainText().length` until cumulative
///    length exceeds targetOffset — that paragraph owns the offset.
/// 3. Compute the caret position within the owning paragraph via
///    `getOffsetForCaret(TextPosition(offset: localOffset), Rect.zero)`.
/// 4. Translate paragraph-local caret to scrollable-local via
///    `getTransformTo(scrollable.findRenderObject())`.
/// 5. `animateTo(y)` on scrollable.position.
///
/// Edge cases:
/// - targetOffset out of range: silent return (no scroll).
/// - Owner paragraph not found (e.g., all content hidden): silent return.
/// - Caller responsible for ensuring [scrollable] has been laid out —
///   typically call after `await WidgetsBinding.instance.endOfFrame`.
class ScrollToOffsetService {
  static Future<void> scrollToCharacterOffset({
    required BuildContext context,
    required ScrollableState scrollable,
    required int targetOffset,
    Duration duration = const Duration(milliseconds: 300),
  }) async {
    final scrollRenderObj = scrollable.context.findRenderObject();
    if (scrollRenderObj == null) return;

    final paragraphs = _collectRenderParagraphs(scrollRenderObj);
    if (paragraphs.isEmpty) return;

    int cumulative = 0;
    RenderParagraph? owner;
    int localOffset = 0;
    for (final p in paragraphs) {
      final len = p.text.toPlainText().length;
      if (cumulative + len > targetOffset) {
        owner = p;
        localOffset = targetOffset - cumulative;
        break;
      }
      cumulative += len;
    }
    if (owner == null) return; // out of range

    final caretLocal = owner.getOffsetForCaret(
      TextPosition(offset: localOffset),
      Rect.zero,
    );
    final transform = owner.getTransformTo(scrollRenderObj);
    final caretInScroll = MatrixUtils.transformPoint(transform, caretLocal);

    await scrollable.position.animateTo(
      caretInScroll.dy,
      duration: duration,
      curve: Curves.easeInOut,
    );
  }

  /// Collect [RenderParagraph] descendants in document order.
  static List<RenderParagraph> _collectRenderParagraphs(RenderObject root) {
    final result = <RenderParagraph>[];
    void visit(RenderObject obj) {
      if (obj is RenderParagraph) {
        result.add(obj);
      }
      obj.visitChildren(visit);
    }
    visit(root);
    return result;
  }
}
```
