# M4.8 EPUB TextPainter Pagination Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace fwfh-based paged EPUB rendering with a TextPainter-driven pipeline (HTML → InlineSpan → line-level pagination → SelectableText.rich), eliminating the -47px drift that causes "last paragraph eaten" overflow at page bottoms.

**Architecture:** Single-source-of-truth rendering — what measures (TextPainter) is what renders (SelectableText.rich). New components: HtmlToInlineSpanConverter (HTML→InlineSpan), LineMetricsPaginator (line-level breaks via computeLineMetrics), InlineSpanSubtreeSlicer (per-page span subtree extraction), SelectablePageView (PageView + SelectableText.rich, dual-page spread aware). All existing infrastructure preserved: M3.3 Locator (chapterIndex, charOffset), SelectionArea, LXGWNeoXiHei font, HighlightRange, paginator.dart PageBreak.

**Tech Stack:** Flutter 3.x, Dart 3.x, `html` package (DOM parsing), `TextPainter` + `computeLineMetrics`, `SelectableText.rich`, `WidgetSpan` (inline images).

**Spec:** `docs/superpowers/specs/2026-07-07-m4.8-epub-textpainter-pagination-design.md`

**Estimated effort:** 15-18 days across 7 phases.

```dart
  required String chapterHtml,
  required TextStyle baseStyle,
  required double imageMaxHeight,
  required double indentEm,
  required bool indentEnabled,
  required List<HighlightRange> highlights,
}) {
  if (chapterHtml.isEmpty) {
    return ConvertedSpans(
      root: TextSpan(text: '', style: baseStyle),
      plainText: '',
      paragraphRanges: const [],
      images: const [],
    );
  }
  final doc = html_parser.HtmlParser(chapterHtml).parse();
  for (final e in doc.querySelectorAll('script, style')) {
    e.remove();
  }
  final buf = StringBuffer();

  final images = <InlineImage>[];
  final children = <InlineSpan>[];

  final blockTags = const {'p', 'div'};
  final blockElements = <dom.Element>[];
  for (final child in doc.body?.nodes ?? const <dom.Node>[]) {
    if (child is dom.Element && blockTags.contains(child.localName)) {
      blockElements.add(child);
    }
  }

  for (var i = 0; i < blockElements.length; i++) {
    final el = blockElements[i];
    final start = buf.length;
    final text = el.text; // plain text of <p>...</p>
    buf.write(text);
    final end = buf.length;
    ranges.add(ParaRange(start, end));
    children.add(TextSpan(text: text, style: baseStyle));
    if (i < blockElements.length - 1) {
      buf.write('\n\n');
      children.add(const TextSpan(text: '\n\n'));
    }
  }

  return ConvertedSpans(
    root: TextSpan(style: baseStyle, children: children),
    plainText: buf.toString(),
    paragraphRanges: ranges,
    images: images,
  );
```

Replace the body of `convert` (after the `if (chapterHtml.isEmpty)` early return) with a recursive walker. Add a private method:

```dart
ConvertedSpans convert({
  required String chapterHtml,
  required TextStyle baseStyle,
  required double imageMaxHeight,
  required double indentEm,
  required bool indentEnabled,
  required List<HighlightRange> highlights,
}) {
  if (chapterHtml.isEmpty) {
    return ConvertedSpans(
      root: TextSpan(text: '', style: baseStyle),
      plainText: '',
      paragraphRanges: const [],
      images: const [],
    );
  }
  final doc = html_parser.HtmlParser(chapterHtml).parse();
  for (final e in doc.querySelectorAll('script, style')) {
    e.remove();
  }
  final buf = StringBuffer();
  final ranges = <ParaRange>[];
  final images = <InlineImage>[];
  final children = <InlineSpan>[];

  final blockTags = const {'p', 'div'};
  final bodyChildren = doc.body?.nodes ?? const <dom.Node>[];
  final blockElements = <dom.Element>[
    for (final n in bodyChildren)
      if (n is dom.Element && blockTags.contains(n.localName)) n,
  ];

  for (var i = 0; i < blockElements.length; i++) {
    final el = blockElements[i];
    final start = buf.length;
    final paraChildren = <InlineSpan>[];
    _walkInline(el, baseStyle, buf, paraChildren, images);
    final end = buf.length;
    ranges.add(ParaRange(start, end));
    children.add(TextSpan(style: baseStyle, children: paraChildren));
    if (i < blockElements.length - 1) {
      buf.write('\n\n');
      children.add(const TextSpan(text: '\n\n'));
    }
  }

  return ConvertedSpans(
    root: TextSpan(style: baseStyle, children: children),
    plainText: buf.toString(),
    paragraphRanges: ranges,
    images: images,
  );
}

void _walkInline(
  dom.Node node,
  TextStyle style,
  StringBuffer buf,
  List<InlineSpan> out,
  List<InlineImage> images,
) {
  if (node is dom.Text) {
    final text = node.text;
    if (text.isNotEmpty) {
      buf.write(text);
      out.add(TextSpan(text: text, style: style));
    }
    return;
  }
  if (node is! dom.Element) return;
  final tag = node.localName;
  final childStyle = _inlineStyle(style, tag);
  final childOut = <InlineSpan>[];
  for (final child in node.nodes) {
    _walkInline(child, childStyle, buf, childOut, images);
  }
  if (childOut.isNotEmpty) {
    if (childOut.length == 1) {
      out.add(childOut.single);
    } else {
      out.add(TextSpan(style: childStyle, children: childOut));
    }
  }
}

TextStyle _inlineStyle(TextStyle base, String? tag) {
  switch (tag) {
    case 'em':
    case 'i':
      return base.copyWith(fontStyle: FontStyle.italic);
    case 'strong':
    case 'b':
      return base.copyWith(fontWeight: FontWeight.bold);
    case 'u':
      return base.copyWith(decoration: TextDecoration.combine(
          [base.decoration ?? TextDecoration.none, TextDecoration.underline]));
    case 'a':
      return base.copyWith(
          color: Colors.blue,
          decoration: TextDecoration.combine(
              [base.decoration ?? TextDecoration.none, TextDecoration.underline]));
    case 'code':
    case 'pre':
      return base.copyWith(fontFamily: 'monospace');
    default:
      return base;
  }
}
```

```dart
  for (final n in bodyChildren)

  final el = blockElements[i];
  final tag = el.localName;
  final start = buf.length;

  // Prefix for list items.
  String? prefix;
  TextStyle? headingStyle;
  if (tag == 'li') {
    // Determine parent <ol>/<ul> and index.
    final parent = el.parent;
    if (parent?.localName == 'ol') {
      final siblings = parent!.querySelectorAll('li');
      final idx = siblings.indexOf(el) + 1;
      prefix = '$idx. ';
    } else {
      prefix = '• ';
    }
  } else if (tag != null && tag.startsWith('h')) {
    final level = int.tryParse(tag.substring(1)) ?? 1;
    final sizeScale = switch (level) {
      1 => 1.5, 2 => 1.3, 3 => 1.15, 4 => 1.0, 5 => 0.95, _ => 0.85,
    };
    headingStyle = baseStyle.copyWith(
      fontWeight: FontWeight.bold,
      fontSize: baseStyle.fontSize! * sizeScale,
    );
  }

  if (prefix != null) {
    buf.write(prefix);
  }
  final effectiveStyle = headingStyle ?? baseStyle;
  final paraChildren = <InlineSpan>[];
  if (prefix != null) {
    paraChildren.add(TextSpan(text: prefix, style: effectiveStyle));
  }
  _walkInline(el, effectiveStyle, buf, paraChildren, images);
  final end = buf.length;
  ranges.add(ParaRange(start, end));
  children.add(TextSpan(style: effectiveStyle, children: paraChildren));
  if (i < blockElements.length - 1) {
    buf.write('\n\n');
    children.add(const TextSpan(text: '\n\n'));
```

```dart
  // rgba(255,235,59,1) or #RRGGBB
  if (rgba.startsWith('#')) {

  }
  final m = RegExp(r'rgba?\(([^)]+)\)').firstMatch(rgba);

  final parts = m.group(1)!.split(',').map((s) => s.trim()).toList();

  return Color.fromARGB((a * 255).round(), r, g, b);
}
```

```dart
InlineSpan _rewriteWithHighlights(
    InlineSpan span, String plainText,
    List<(int, int, Color)> ranges, [int offset = 0]) {
  if (span is TextSpan) {
    final text = span.text ?? '';
    final newChildren = <InlineSpan>[];
    var i = 0;
    while (i < text.length) {
      final absStart = offset + i;
      final hit = ranges.where((r) => r.$1 <= absStart && absStart < r.$2).toList();
      int chunkEnd;
      Color? bg;
      if (hit.isEmpty) {
        // No highlight; extend until next highlight start
        final nextStarts = ranges.map((r) => r.$1).where((s) => s > absStart).toList();
        chunkEnd = nextStarts.isEmpty ? text.length : min(nextStarts.reduce(min), text.length);
      } else {
        bg = hit.first.$3;
        chunkEnd = min(hit.first.$2 - absStart + i, text.length);
      }
      newChildren.add(TextSpan(
        text: text.substring(i, chunkEnd),
        style: span.style?.copyWith(backgroundColor: bg),
      ));
      i = chunkEnd;
    }
    if (span.children != null) {
      var childOffset = offset + text.length;
      for (final child in span.children!) {
        newChildren.add(_rewriteWithHighlights(child, plainText, ranges, childOffset));
        childOffset += _spanTextLength(child);
      }
    }
    return TextSpan(style: span.style, children: newChildren);
  }
  return span;
}

int _spanTextLength(InlineSpan span) {
  if (span is TextSpan) {
    var len = (span.text ?? '').length;
    if (span.children != null) {
      for (final c in span.children!) {
        len += _spanTextLength(c);
      }
    }
    return len;
  }
  if (span is WidgetSpan) return 1;
  return 0;
}
```

```dart
// SPDX-License-Identifier: AGPL-3.0-or-later
// Copyright (C) 2026 ZhiYue Open Source contributors

import 'package:flutter/painting.dart';
import 'package:reader_engine/core/page_break.dart';

/// Line-level EPUB paginator using [TextPainter.computeLineMetrics].
///
/// Core invariant: the same [InlineSpan] passed to this paginator is also
/// rendered by `SelectableText.rich`, so measurement and rendering share
/// one layout engine → drift = 0 (see spec §3.2).
class LineMetricsPaginator {
  const LineMetricsPaginator();

  List<PageBreak> paginate({
    required InlineSpan span,
    required String plainText,
    required Size viewport,
    required double marginPx,
    required double lineHeight,
  }) {
    final usableWidth = (viewport.width - 2 * marginPx).clamp(50.0, double.infinity);
    final usableHeight = viewport.height - 2 * marginPx;
    final safetyMargin = 0.3 * lineHeight;

    final tp = TextPainter(
      text: span,
      textDirection: TextDirection.ltr,
    )..layout(maxWidth: usableWidth);

    if (plainText.isEmpty) {
      tp.dispose();
      return const [PageBreak(startOffset: 0, endOffset: 0)];
    }

    final lines = tp.computeLineMetrics();
    final breaks = <PageBreak>[];
    var pageStart = 0;
    var pageHeight = 0.0;

    for (final line in lines) {
      final lineTop = line.baseline - line.ascent;
      final startOffset = tp.getPositionForOffset(Offset(0, lineTop)).offset;
      // If adding this line would overflow and we already have content on the page:
      if (pageHeight + line.height > usableHeight - safetyMargin && pageHeight > 0) {
        breaks.add(PageBreak(startOffset: pageStart, endOffset: startOffset));
        pageStart = startOffset;
        pageHeight = line.height;
      } else {
        pageHeight += line.height;
      }
    }
    breaks.add(PageBreak(startOffset: pageStart, endOffset: plainText.length));
    tp.dispose();
    return breaks;
  }
}
```

```dart
// SPDX-License-Identifier: AGPL-3.0-or-later
// Copyright (C) 2026 ZhiYue Open Source contributors

import 'package:flutter/widgets.dart';

/// Extracts a sub-[InlineSpan] tree covering [start, end) character range
/// of the input span's plain text. Used to slice a per-page InlineSpan from
/// a chapter-level InlineSpan for [SelectableText.rich] rendering.
class InlineSpanSubtreeSlicer {
  const InlineSpanSubtreeSlicer();

  static InlineSpan slice(InlineSpan root, int start, int end) {
    if (start >= end) return const TextSpan(text: '');
    final builder = _SliceBuilder(start, end);
    builder.visit(root, 0);
    return TextSpan(style: _styleOf(root), children: builder.out);
  }

  static TextStyle? _styleOf(InlineSpan span) {
    if (span is TextSpan) return span.style;
    return null;
  }
}

class _SliceBuilder {
  final int start;
  final int end;
  final List<InlineSpan> out = <InlineSpan>[];

  _SliceBuilder(this.start, this.end);

  /// Visits [span] given its text starts at [offset] in plainText.
  /// Returns the plainText length consumed.
  int visit(InlineSpan span, int offset) {
    if (span is TextSpan) {
      return _visitTextSpan(span, offset);
    }
    if (span is WidgetSpan) {
      return _visitWidgetSpan(span, offset);
    }
    return 0;
  }

  int _visitTextSpan(TextSpan span, int offset) {
    var consumed = 0;
    final text = span.text ?? '';
    if (text.isNotEmpty) {
      final absEnd = offset + text.length;
      if (absEnd > start && offset < end) {
        final localStart = (start - offset).clamp(0, text.length);
        final localEnd = (end - offset).clamp(0, text.length);
        if (localEnd > localStart) {
          out.add(TextSpan(
            text: text.substring(localStart, localEnd),
            style: span.style,
          ));
        }
      }
      consumed = text.length;
    }
    if (span.children != null) {
      var childOffset = offset + consumed;
      for (final child in span.children!) {
        childOffset += visit(child, childOffset);
      }
      consumed = childOffset - offset;
    }
    return consumed;
  }

  int _visitWidgetSpan(WidgetSpan span, int offset) {
    // WidgetSpan occupies 1 char (U+FFFC).
    if (offset >= start && offset < end) {
      out.add(span);
    }
    return 1;
  }
}
```

```dart
import 'package:flutter/material.dart';
import 'package:reader_engine/core/page_break.dart';
import 'package:reader_engine_epub_view/inline_span_subtree_slicer.dart';

/// Renders [breaks] pages of [rootSpan] using a [PageView] of
/// [SelectableText.rich]. Each page item slices the chapter-level
/// [rootSpan] to its break range via [InlineSpanSubtreeSlicer].
class SelectablePageView extends StatelessWidget {
  const SelectablePageView({
    super.key,
    required this.rootSpan,
    required this.plainText,
    required this.breaks,
    required this.dualPageMode,
    this.initialPage = 0,
    this.controller,
    this.onPageChanged,
    this.gutterPx = 24,
  });

  final InlineSpan rootSpan;
  final String plainText;
  final List<PageBreak> breaks;
  final bool dualPageMode;
  final int initialPage;
  final PageController? controller;
  final ValueChanged<int>? onPageChanged;
  final double gutterPx;

  @override
  Widget build(BuildContext context) {
    final itemCount = dualPageMode
        ? ((breaks.length + 1) ~/ 2)
        : breaks.length;
    return PageView.builder(
      controller: controller ?? PageController(initialPage: initialPage),
      itemCount: itemCount,
      onPageChanged: onPageChanged,
      itemBuilder: (ctx, spreadIdx) {
        if (dualPageMode) {
          return _buildSpread(spreadIdx);
        }
        return _buildSinglePage(spreadIdx);
      },
    );
  }

  Widget _buildSinglePage(int idx) {
    final b = breaks[idx];
    final span = InlineSpanSubtreeSlicer.slice(rootSpan, b.startOffset, b.endOffset);
    return Padding(
      padding: const EdgeInsets.all(16),
      child: SelectableText.rich(span),
    );
  }

  Widget _buildSpread(int spreadIdx) {
    final leftIdx = spreadIdx * 2;
    final rightIdx = leftIdx + 1;
    final left = _buildSinglePage(leftIdx);
    final right = rightIdx < breaks.length
        ? Expanded(child: _buildSinglePage(rightIdx))
        : const Expanded(child: SizedBox.shrink());
    return Row(
      crossAxisAlignment: CrossAxisAlignment.stretch,
      children: [
        Expanded(child: left),
        SizedBox(width: gutterPx),
        right,
      ],
    );
  }
}
```
