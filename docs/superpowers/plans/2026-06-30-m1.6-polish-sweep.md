```dart
  /// Construct a [Book] from a drift-generated [BookData] row. Centralizes
  /// the row → domain model mapping so call sites don't each have to know
  /// the 11-field layout. The drift schema is the source of truth: if a
  /// column is added there, regenerating drift + adding it here updates
  /// every read site.
  factory Book.fromRow(BookData row) {
    return Book(
      id: row.id,
      title: row.title,
      author: row.author,
      format: row.format == 'epub' ? BookFormat.epub : BookFormat.pdf,
      filePath: row.filePath,
      fileHash: row.fileHash,
      fileSizeBytes: row.fileSizeBytes,
      coverPath: row.coverPath,
      addedAt: row.addedAt,
      lastReadAt: row.lastReadAt,
      sortOrder: row.sortOrder,
    );
  }
```

```dart
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:http/http.dart' as http;

/// Process-wide shared [http.Client]. Reused across AI services
/// ([OpenAiCompatibleClient], [AiGuide]) and the WebDAV sync client for
/// connection pooling. Override in tests with a mock client.
final httpClientProvider = Provider<http.Client>((ref) {
  final client = http.Client();
  ref.onDispose(client.close);
  return client;
});
```

```cpp
#include "deep_link_parser.h"

#include <algorithm>
#include <cctype>

namespace zhiyue {

const wchar_t* kZhiYueWindowProp = L"ZhiYueWindowProp";
const ULONG_PTR kCopyDataArgvTag = 1;
const wchar_t* kZhiYueMutexName = L"Local\\ZhiYueApp";

std::string WideToUtf8(const std::wstring& w) {
  if (w.empty()) return std::string();
  int needed = WideCharToMultiByte(CP_UTF8, 0, w.c_str(),
                                   static_cast<int>(w.size()),
                                   nullptr, 0, nullptr, nullptr);
  std::string out(needed, '\0');
  WideCharToMultiByte(CP_UTF8, 0, w.c_str(),
                      static_cast<int>(w.size()),
                      out.data(), needed, nullptr, nullptr);
  return out;
}

COPYDATASTRUCT BuildCopyData(const std::string& argv) {
  COPYDATASTRUCT cds = {};
  cds.dwData = kCopyDataArgvTag;
  cds.cbData = static_cast<DWORD>(argv.size() + 1);
  cds.lpData = const_cast<char*>(argv.c_str());
  return cds;
}

bool IsBookFileArg(const std::string& arg) {
  auto dot_pos = arg.find_last_of('.');
  if (dot_pos == std::string::npos) return false;
  std::string ext = arg.substr(dot_pos);
  std::transform(ext.begin(), ext.end(), ext.begin(),
                 [](unsigned char c) { return std::tolower(c); });
  return ext == ".epub" || ext == ".pdf";
}

}  // namespace zhiyue
```

````markdown
## 8. 常见 bug 修复

### 8.1 `Stream.timeout` 不触发（async* 源）

**症状**：异步生成器（`async*`）yield 后 await 一个永不完成的 future，`Stream.timeout` 不触发。

**原因**：Dart 的 `Stream.timeout` 在源订阅 paused 时不走表。async* 源 yield 后 await 触发 paused 状态。

**修复**：用 `StreamIterator.moveNext().timeout()` 替代，无 pause 耦合。见 `app/lib/services/ai/book_qa_service.dart` 的实现。

### 8.2 drift DAO mocking 困难

**症状**：drift DAO 直接 fake 麻烦（`DatabaseAccessor` mixin 要求 db accessor）。

**修复**：抽 `Like` 接口。例如 `QaHistoryDaoLike`，让 widget test 用 fake 实现。见 `app/lib/data/local/daos/qa_history_dao.dart`。

### 8.3 provider 默认 throw UnimplementedError 引发 widget test 噪音

**症状**：widget test 不 override 某 provider 时，`initState` post-frame 读它会抛 `UnimplementedError`，被 unawaited future 吞但打印 stack trace。

**修复**：默认值改 NoOp（inert）+ production override 在 `app/lib/core/app_providers.dart` 的 `providers` list。见 `_NoOpQaHistoryDao` 模式。

### 8.4 `ActionChip.onLongPress` 不存在

**症状**：Flutter `ActionChip` 没有 `onLongPress` 回调。

**修复**：用 `GestureDetector(behavior: HitTestBehavior.translucent, onLongPress: ...)` 包，保留 chip 的 onPressed 涟漪。

### 8.5 `TapGestureRecognizer` 不自动 dispose

**症状**：`TextSpan.recognizer` 创建的 recognizer framework 不自动 dispose，造成内存泄漏。

**修复**：StatefulWidget 持有 refs，在 `dispose()` 和 `didUpdateWidget` 中手动 dispose。见 `app/lib/features/ai/widgets/qa_answer_view.dart`。
````
