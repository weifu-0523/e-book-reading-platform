# M3.5 — Settings + Fonts + i18n + Multi-profile Typography Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把 M3.4 留下的"无设置页 + 排版不持久化 + 中文硬编码 + 8 项 follow-up"一次性升级为"集中设置页 + 多 profile 持久化排版 + 用户可扩展字体库 + 中英双语 + App 主题切换"。

**Architecture:** 沿用现有 `features/<domain>/` 模式，新增 `features/settings/` 模块集中管理用户配置入口（profile/fonts/locale/theme）。i18n 用 flutter_gen_l10n + arb（pubspec 已含 flutter_localizations + intl 0.20）。字体持久化用新增 `user_fonts` drift 表（schema v5）+ FileStorage 的 fontsDir。多 profile 激活已有的 `typography_profiles` 表，加 `setDefault` 事务。

**Tech Stack:** Flutter 3.44 / Dart 3.12 / Riverpod 2.6 / drift 2.18 / go_router 14.2 / file_picker 8.1.2 / crypto 3.0.5 / uuid 4.4.2 / intl 0.20 / flutter_localizations。

    final en = _loadKeys('lib/l10n/app_en.arb');
    expect(en.keys.toSet(), equals(zh.keys.toSet()),
        reason: 'zh and en arb must have identical key sets');
  });

  test('critical keys present in both arb files', () {
    for (final arb in ['lib/l10n/app_zh.arb', 'lib/l10n/app_en.arb']) {
      final keys = _loadKeys(arb).keys.toSet();
      for (final required in [
        'appTitle',
        'settingsTitle',
        'settingsSectionReading',
        'settingsSectionFonts',
        'settingsSectionAppearance',
        'settingsSectionLanguage',
        'settingsSectionAbout',
      ]) {
        expect(keys, contains(required),
            reason: '$arb missing required key: $required');
      }
    }
  });
}

Map<String, dynamic> _loadKeys(String relPath) {
  final file = File(relPath);
  final raw = file.readAsStringSync();
  final decoded = jsonDecode(raw) as Map<String, dynamic>;
  // 过滤掉 @metadata key (@@locale 等)
  return Map.fromEntries(
      decoded.entries.where((e) => !e.key.startsWith('@')));
}

// app/lib/features/settings/providers/locale_provider.dart
import 'package:flutter_riverpod/flutter_riverpod.dart';

import '../../../core/providers.dart' show settingsRepositoryProvider;
import '../../../data/repositories/settings_repository.dart';
import '../models/locale_setting.dart';

final localeProvider =
    StateNotifierProvider<LocaleNotifier, LocaleSetting>((ref) {
  return LocaleNotifier(ref.watch(settingsRepositoryProvider));
});

class LocaleNotifier extends StateNotifier<LocaleSetting> {
  LocaleNotifier(this._repo) : super(LocaleSetting.system) {
    _load();
  }

  final SettingsRepository _repo;

  Future<void> _load() async {
    final name = await _repo.get<String?>('app.locale', null);
    state = LocaleSetting.fromName(name);
  }

  Future<void> setLocale(LocaleSetting v) async {
    state = v;
    await _repo.set<String?>('app.locale', v.name);
  }
}

      (delete(typographyProfiles)..where((t) => t.id.equals(id))).go();

    required String id,

    String? settingsJson,
    bool? isDefault,
  }) {
    return (update(typographyProfiles)..where((t) => t.id.equals(id))).write(
      TypographyProfilesCompanion(
        name: name == null ? const Value.absent() : Value(name),
        settings: settingsJson == null ? const Value.absent() : Value(settingsJson),

// app/lib/data/repositories/typography_profile_repository.dart
import 'dart:convert';

import 'package:drift/drift.dart';

import '../local/database.dart';
import 'package:reader_engine/core/typography_settings.dart';
import '../data/models/typography_profile.dart' as model;

class TypographyProfileRepository {
  TypographyProfileRepository(this._db);

  final AppDatabase _db;

  Future<List<model.TypographyProfile>> getAll() async {
    final rows = await _db.typographyDao.getAllProfiles();
    return rows
        .map((r) => model.TypographyProfile(
              id: r.id,
              name: r.name,
              settings: TypographySettings.fromJson(
                  jsonDecode(r.settings) as Map<String, dynamic>),
              isDefault: r.isDefault,
            ))
        .toList();
  }

  Future<model.TypographyProfile?> getDefault() async {
    final all = await getAll();
    return all.firstWhere(
      (p) => p.isDefault,
      orElse: () => all.first,
    );
  }

  Future<model.TypographyProfile?> getById(String id) async {
    final rows = await _db.typographyDao.getAllProfiles();
    final r = rows.firstWhere((x) => x.id == id);
    return model.TypographyProfile(
      id: r.id,
      name: r.name,
      settings: TypographySettings.fromJson(
          jsonDecode(r.settings) as Map<String, dynamic>),
      isDefault: r.isDefault,
    );
  }

  Future<model.TypographyProfile> seedDefaultIfEmpty() async {
    final all = await getAll();
    if (all.isNotEmpty) return all.first;
    return insert(
      name: '默认',
      settings: TypographySettings.defaults(),
      isDefault: true,
    );
  }

  Future<model.TypographyProfile> insert({
    required String name,
    required TypographySettings settings,
    required bool isDefault,
  }) async {
    final id = _uuid();
    await _db.typographyDao.insertProfile(
      id: id,
      name: name,
      settingsJson: jsonEncode(settings.toJson()),
      isDefault: isDefault,
    );
    return model.TypographyProfile(
        id: id, name: name, settings: settings, isDefault: isDefault);
  }

  Future<void> update(String id,
      {String? name, TypographySettings? settings, bool? isDefault}) async {
    await _db.typographyDao.updateProfile(
      id: id,
      name: name,
      settingsJson: settings == null ? null : jsonEncode(settings.toJson()),
      isDefault: isDefault,
    );
  }

  Future<void> setDefault(String id) => _db.typographyDao.setDefault(id);

  Future<void> delete(String id) => _db.typographyDao.deleteProfile(id);

  Future<List<model.TypographyProfile>> findProfilesReferencingFont(
      String familyName) async {
    final all = await getAll();
    return all.where((p) => p.settings.fontFamily == familyName).toList();
  }

  String _uuid() =>
      DateTime.now().microsecondsSinceEpoch.toRadixString(36) +
      (idCounter++).toRadixString(36);
  static int idCounter = 0;
}

class TypographyProfilesPage extends ConsumerWidget {
  const TypographyProfilesPage({super.key});

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final profilesAsync = ref.watch(typographyProfileProvider);
    return Scaffold(
      appBar: AppBar(
        title: Text(AppLocalizations.of(context)!.settingsTypographyProfiles),
        actions: [
          IconButton(
            icon: const Icon(Icons.add),
            onPressed: () => _showCreateDialog(context, ref),
          ),
        ],
      ),
      body: profilesAsync.when(
        loading: () => const Center(child: CircularProgressIndicator()),
        error: (e, _) => Center(child: Text('$e')),
        data: (profiles) => profiles.isEmpty
            ? const Center(child: Text('无预设'))
            : ListView(
                children: profiles
                    .map((p) => ProfileTile(
                          profile: p,
                          isOnly: profiles.length == 1,
                          onSetDefault: () => ref
                              .read(typographyProfileRepositoryProvider)
                              .setDefault(p.id)
                              .then((_) => ref.invalidate(
                                  typographyProfileProvider)),
                          onRename: () => _showRenameDialog(context, ref, p),
                          onDelete: () => _showDeleteConfirm(context, ref, p),
                        ))
                    .toList(),
              ),
      ),
    );
  }

  void _showCreateDialog(BuildContext context, WidgetRef ref) {
    final ctrl = TextEditingController(text: '新预设');
    showDialog<void>(
      context: context,
      builder: (ctx) => AlertDialog(
        title: const Text('新建预设'),
        content: TextField(
          controller: ctrl,
          decoration: const InputDecoration(labelText: '名称'),
          autofocus: true,
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(ctx),
            child: const Text('取消'),
          ),
          FilledButton(
            onPressed: () async {
              final name = ctrl.text.trim();
              if (name.isEmpty) return;
              Navigator.pop(ctx);
              final repo = ref.read(typographyProfileRepositoryProvider);
              await repo.insert(
                name: name,
                settings: await repo.getDefault().then((p) => p!.settings),
                isDefault: false,
              );
              ref.invalidate(typographyProfileProvider);
            },
            child: const Text('创建'),
          ),
        ],
      ),
    );
  }

  void _showRenameDialog(BuildContext context, WidgetRef ref, TypographyProfile p) {
    final ctrl = TextEditingController(text: p.name);
    showDialog<void>(
      context: context,
      builder: (ctx) => AlertDialog(
        title: const Text('重命名'),
        content: TextField(
          controller: ctrl,
          decoration: const InputDecoration(labelText: '名称'),
          autofocus: true,
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(ctx),
            child: const Text('取消'),
          ),
          FilledButton(
            onPressed: () async {
              final name = ctrl.text.trim();
              if (name.isEmpty) return;
              Navigator.pop(ctx);
              await ref.read(typographyProfileRepositoryProvider).update(p.id, name: name);
              ref.invalidate(typographyProfileProvider);
            },
            child: const Text('保存'),
          ),
        ],
      ),
    );
  }

  void _showDeleteConfirm(BuildContext context, WidgetRef ref, TypographyProfile p) {
    showDialog<void>(
      context: context,
      builder: (ctx) => AlertDialog(
        title: const Text('删除预设'),
        content: Text('确定删除「${p.name}」？'),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(ctx),
            child: const Text('取消'),
          ),
          FilledButton(
            style: FilledButton.styleFrom(backgroundColor: Colors.red),
            onPressed: () async {
              Navigator.pop(ctx);
              await ref.read(typographyProfileRepositoryProvider).delete(p.id);
              ref.invalidate(typographyProfileProvider);
            },
            child: const Text('删除'),
          ),
        ],
      ),
    );
  }
}

### Step 4.10: Create `FontImportService`

```dart
// app/lib/features/settings/services/font_import_service.dart
import 'dart:io';

import 'package:crypto/crypto.dart';
import 'package:file_picker/file_picker.dart';
import 'package:flutter/services.dart' show FontLoader, rootBundle;
import 'package:path/path.dart' as p;
import 'package:uuid/uuid.dart';

import '../../../data/models/user_font.dart';
import '../../../data/repositories/user_font_repository.dart';

import '../models/font_import_result.dart';

class FontImportService {
  FontImportService(this._repo, this._fontsDir);

  final UserFontRepository _repo;
  final Directory _fontsDir;

  Future<FontImportResult> importFont() async {
    final pick = await FilePicker.platform.pickFiles(
      type: FileType.custom,
      allowedExtensions: const ['ttf', 'otf', 'TTF', 'OTF'],
    );
    if (pick == null || pick.files.isEmpty) return const FontImportCancelled();

    final pickedPath = pick.files.single.path;
    if (pickedPath == null) return const FontImportReadError('path null');

    final bytes = await _readBytes(pickedPath);

    final hash = sha1.convert(bytes).toString();
    final existing = await _repo.findByHash(hash);
    if (existing != null) return FontImportAlreadyExists(existing);

    final originalName = pick.files.single.name;
    final safeName = _sanitize(originalName);
    final stamp = DateTime.now().microsecondsSinceEpoch;
    final destPath = p.join(_fontsDir.path, '${stamp}_$safeName');
    final destFile = File(destPath);
    await destFile.writeAsBytes(bytes);

    final familyName = p.basenameWithoutExtension(destPath);
    final loader = FontLoader(familyName)..addFont(FileAsset(path: destPath));
    try {
      await loader.load();
    } catch (e) {
      await destFile.delete();
      return FontImportInvalid('FontLoader.load failed: $e');
    }

    final font = UserFont(
      id: const Uuid().v4(),
      displayName: familyName,
      fileName: p.basename(destPath),
      familyName: familyName,
      hashSha1: hash,
      absolutePath: destPath,
      addedAt: DateTime.now(),
    );
    await _repo.insert(font);
    return FontImportSuccess(font);
  }

  Future<List<int>> _readBytes(String path) async {
    try {
      return await File(path).readAsBytes();
    } catch (e) {
      throw Exception('readAsBytes failed: $e');
    }
  }

  String _sanitize(String name) {
    final cleaned = name.replaceAll(RegExp(r'[^\w.\-]'), '_');
    return cleaned.isEmpty ? 'font.ttf' : cleaned;
  }
}

extension on Future<FontImportResult> {}
```

**Note:** The `FontLoader.load()` will throw on corrupted font files in most platforms. Test fixtures need to verify this behavior on Windows/Linux test runner.

    final service = await ref.read(fontImportServiceFutureProvider.future);
    FontImportResult result;
    try {
      result = await service.importFont();
    } catch (e) {
      messenger.showSnackBar(SnackBar(content: Text('导入失败：$e')));
      return;

    switch (result) {
      case FontImportCancelled():
        break;

      case FontImportSuccess(:final font):
        ref.invalidate(fontLibraryProvider);
