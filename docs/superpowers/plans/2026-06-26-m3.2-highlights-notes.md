class EpubRangeLocator extends Locator {
  final int chapterIndex;
  final String? chapterTitle;
  final int startOffset;
  final int endOffset;
  final String quote;

  const EpubRangeLocator({
    required this.chapterIndex,
    required this.startOffset,
    required this.endOffset,
    required this.quote,
    this.chapterTitle,
  })  : assert(startOffset >= 0),
        assert(endOffset > startOffset),
        assert(quote.isNotEmpty);

  @override
  Map<String, dynamic> toJson() => {
        'type': 'epubRange',
        'chapterIndex': chapterIndex,
        if (chapterTitle != null) 'chapterTitle': chapterTitle,
        'startOffset': startOffset,
        'endOffset': endOffset,
        'quote': quote,
      };

  @override
  String toJsonString() => '$chapterIndex:$startOffset-$endOffset';

  @override
  bool operator ==(Object other) =>
      identical(this, other) ||
      other is EpubRangeLocator &&
          chapterIndex == other.chapterIndex &&
          chapterTitle == other.chapterTitle &&
          startOffset == other.startOffset &&
          endOffset == other.endOffset &&
          quote == other.quote;

  @override
  int get hashCode =>
      Object.hash(chapterIndex, chapterTitle, startOffset, endOffset, quote);
}

// reader_engine_epub_view/lib/plain_text_utils.dart
import 'package:meta/meta.dart';

/// HTML 处理工具：plain text 提取 + plain→HTML offset 映射。
/// 供 [HtmlHighlightInjector] 与选区反查共用，保证坐标系一致。
@immutable
class PlainTextUtils {
  const PlainTextUtils._();

  /// 把 HTML 转为 (plainText, htmlOffsets)。
  ///
  /// - plainText：去除所有标签、解码 HTML 实体后的纯文本
  /// - htmlOffsets[i]：plainText 第 i 个字符在原 HTML 中的起始 offset
  ///
  /// 已知边界：
  /// - 属性值含 `<`（`<a title="a<b">`）：按 tag-skip 处理，可能误判；EPUB 罕见
  /// - 字符引用 `&#nn;` 数字形式支持；命名实体仅 &amp; &lt; &gt; &quot; &apos; &nbsp;
  static (String, List<int>) htmlToPlainWithOffsets(String html) {
    final plain = StringBuffer();
    final offsets = <int>[];
    var i = 0;
    while (i < html.length) {
      final ch = html[i];
      if (ch == '<') {
        // tag-skip 直到 '>'
        final end = html.indexOf('>', i);
        if (end < 0) break;
        i = end + 1;
        continue;
      }
      if (ch == '&') {
        // entity 解码
        final semi = html.indexOf(';', i);
        if (semi > 0 && semi - i <= 6) {
          final entity = html.substring(i + 1, semi);
          final decoded = _decodeEntity(entity);
          if (decoded != null) {
            for (var j = 0; j < decoded.length; j++) {
              plain.write(decoded[j]);
              offsets.add(i); // 每个 decoded 字符的 HTML 起点 = '&' 位置
            }
            i = semi + 1;
            continue;
          }
        }
      }
      plain.write(ch);
      offsets.add(i);
      i++;
    }
    return (plain.toString(), offsets);
  }

  static String? _decodeEntity(String entity) {
    switch (entity) {
      case 'amp':
        return '&';
      case 'lt':
        return '<';
      case 'gt':
        return '>';
      case 'quot':
        return '"';
      case 'apos':
        return "'";
      case 'nbsp':
        return ' ';
    }
    if (entity.startsWith('#')) {
      final code = int.tryParse(entity.substring(1));
      if (code != null) return String.fromCharCode(code);
    }
    return null;
  }
}
```

// reader_engine_epub_view/lib/html_highlight_injector.dart
import 'package:reader_engine/core/locator.dart';

/// 用于避免 reader_engine_epub_view 反向依赖 app 包：
/// 注入器只看一个最小化的 Highlight 视图，由调用方从 Annotation 适配。
class HighlightRange {
  final int startOffset;
  final int endOffset;
  final String colorRgba;
  final bool hasNote;
  HighlightRange(this.startOffset, this.endOffset, this.colorRgba,
      {this.hasNote = false});
}

/// 把高亮区间注入 HTML。
///
/// 1. 用 [PlainTextUtils] 把 html → plainText + offsetMap
/// 2. 对每个 HighlightRange，反查 html start/end offset
/// 3. 在 html 对应位置插 `<span style="...">...</span>`
/// 4. 多个高亮重叠时：按 startOffset 升序处理，已注入区间内的后续高亮被截断（log warning）
class HtmlHighlightInjector {
  const HtmlHighlightInjector._();

  static String inject(String html, List<HighlightRange> highlights) {
    if (highlights.isEmpty) return html;
    final (plain, offsetMap) = PlainTextUtils.htmlToPlainWithOffsets(html);
    // 按 startOffset 升序，确保注入位置从前到后
    final sorted = [...highlights]
      ..sort((a, b) => a.startOffset.compareTo(b.startOffset));

    final buf = StringBuffer();
    var cursor = 0; // 当前已写入到 html 的 offset
    var covered = <List<int>>[]; // 已注入的 plain 区间，用于重叠检测

    for (final h in sorted) {
      if (h.startOffset < 0 || h.endOffset > plain.length) continue;
      if (h.endOffset <= h.startOffset) continue;
      // 重叠检测：当前高亮起点 < 已注入任意区间终点 → 截断
      var effStart = h.startOffset;
      var effEnd = h.endOffset;
      for (final c in covered) {
        if (effStart < c[1]) {
          effStart = c[1]; // 截到前一段之后
        }
      }
      if (effStart >= effEnd) continue;

      final htmlStart = offsetMap[effStart];
      // endOffset 是 exclusive；取 endOffset-1 的 html offset +1
      final htmlEnd = offsetMap[effEnd - 1] + 1;

      buf.write(html.substring(cursor, htmlStart));
      final style = _buildStyle(h.colorRgba, h.hasNote);
      buf.write('<span style="$style">');
      buf.write(html.substring(htmlStart, htmlEnd));
      buf.write('</span>');
      cursor = htmlEnd;
      covered.add([effStart, effEnd]);
    }
    buf.write(html.substring(cursor));
    return buf.toString();
  }

  static String _buildStyle(String rgba, bool hasNote) {
    final bg = 'background-color: $rgba';
    if (hasNote) return '$bg; border-bottom: 2px solid $rgba';
    return bg;
  }
}
```

class EpubxRenderer implements EpubRenderer {

      StreamController<HighlightSelection>.broadcast();

  @override
  Future<void> injectHighlights(List<HighlightRange> highlights) async {

      ..addAll(highlights);

  @visibleForTesting
  String currentHighlightedHtml() {
    final chapters = flatChapters;
    if (chapters.isEmpty) return '';

              h.endOffset - chapterStart,
              h.colorRgba,

class FakeNotesRepository implements NotesRepository {
  final Map<String, BookNote> _notes = {};
  final Map<String, List<Annotation>> _annotationsByNote = {};
  final Map<String, StreamController<List<Annotation>>> _controllers = {};

  StreamController<List<Annotation>> _ctlFor(String bookId) {
    return _controllers.putIfAbsent(
        bookId, () => StreamController<List<Annotation>>.broadcast());
  }

  @override
  Stream<List<Annotation>> watchAnnotations(String bookId) async* {
    final note = _notes[bookId];
    yield note == null ? const [] : (_annotationsByNote[note.id] ?? const []);
    yield* _ctlFor(bookId).stream;
  }

  @override
  Future<BookNote?> getNoteByBook(String bookId) async => _notes[bookId];

  @override
  Future<BookNote> createEmptyNote({
    required String id,
    required String bookId,
    required String bookTitle,

    _notes[bookId] = n;
    _annotationsByNote[id] = [];
    return n;
  }

  @override
  Future<void> updateNote(BookNote note) async {
    _notes[note.bookId] = note;
  }

  @override
  Future<List<Annotation>> listAnnotations(String noteId) async =>
      _annotationsByNote[noteId] ?? const [];

  @override
  Future<void> addAnnotation(Annotation a) async {
    _annotationsByNote.putIfAbsent(a.bookNoteId, () => []).add(a);
    final bookId = _notes.entries
        .firstWhere((e) => e.value.id == a.bookNoteId,
            orElse: () => MapEntry(a.bookNoteId, BookNote.empty(
                id: a.bookNoteId, bookId: a.bookNoteId, bookTitle: '')))
        .key;
    _ctlFor(bookId).add(List.unmodifiable(_annotationsByNote[a.bookNoteId]!));
  }

  @override
  Future<void> updateAnnotation(Annotation a) async {
    final list = _annotationsByNote[a.bookNoteId] ?? [];
    final idx = list.indexWhere((x) => x.id == a.id);
    if (idx >= 0) list[idx] = a;
  }

  @override
  Future<void> deleteAnnotation(String id) async {
    for (final list in _annotationsByNote.values) {
      list.removeWhere((x) => x.id == id);
    }
  }

  @override
  Future<List<Annotation>> searchAnnotations(String query,

    filtered = filtered.where((a) => tags.every(a.tags.contains)).toList();

String _sanitizeFtsQuery(String query) {
  return query
      .split(RegExp(r'\s+'))
      .where((s) => s.isNotEmpty)

        final cleaned = s.replaceAll('"', '');
        return cleaned.isEmpty ? null : '"$cleaned"';

/// 标签编辑器：chips + 输入框。输入回车或 done 添加；chip × 删除。
class TagEditor extends StatefulWidget {
  final List<String> tags;
  final List<String> suggestions;
  final ValueChanged<List<String>> onChanged;

  const TagEditor({
    super.key,
    required this.tags,
    required this.suggestions,
    required this.onChanged,
  });

  State<TagEditor> createState() => _TagEditorState();

class _TagEditorState extends State<TagEditor> {
  final _controller = TextEditingController();
  final _focus = FocusNode();

  @override
  void dispose() {
    _controller.dispose();
    _focus.dispose();
    super.dispose();
  }

  void _add(String value) {
    final v = value.trim();
    if (v.isEmpty || widget.tags.contains(v)) {
      _controller.clear();
      return;
    }
    widget.onChanged([...widget.tags, v]);
    _controller.clear();
    _focus.requestFocus();
  }

  void _remove(String value) {
    widget.onChanged(widget.tags.where((t) => t != value).toList());
  }

  @override
  Widget build(BuildContext context) {
    return Wrap(
      spacing: 8,
      runSpacing: 8,
      children: [
        ...widget.tags.map((t) => Chip(
              label: Text(t),
              onDeleted: () => _remove(t),
            )),
        SizedBox(
          width: 120,
          child: Autocomplete<String>(
            optionsBuilder: (textEditingValue) {
              final v = textEditingValue.text;
              if (v.isEmpty) return const Iterable.empty();
              return widget.suggestions.where((s) => s.contains(v));
            },
            fieldViewBuilder: (ctx, controller, focus, onSubmitted) {
              // 用本地 _controller 替换 autocomplete 内部 controller，方便测试
              return TextField(

                focusNode: _focus,

                  isDense: true,

            },
            onSelected: _add,

// app/lib/features/reader/widgets/notes_drawer.dart
import 'package:flutter/material.dart';
import '../../../data/models/annotation.dart';
import 'annotation_tile.dart';

class NotesDrawer extends StatelessWidget {
  final List<Annotation> annotations;
  final Map<int, String> chapterTitles;
  final ValueChanged<String> onTap;
  final ValueChanged<String> onLongPress;
  final VoidCallback onClose;

  const NotesDrawer({
    super.key,
    required this.annotations,
    required this.chapterTitles,
    required this.onTap,
    required this.onLongPress,
    required this.onClose,
  });

  @override
  Widget build(BuildContext context) {
    if (annotations.isEmpty) {
      return Drawer(
        child: SafeArea(
          child: Column(
            children: [
              _buildHeader(),
              const Expanded(child: Center(child: Text('本书暂无高亮与笔记'))),
            ],
          ),
        ),
      );
    }
    // 按 chapterIndex 分组，每组内按 updatedAt 倒序
    final groups = <int, List<Annotation>>{};
    for (final a in annotations) {
      final locator = a.locator;
      if (locator is EpubRangeLocator) {
        groups.putIfAbsent(locator.chapterIndex, () => []).add(a);
      }
    }
    final sortedKeys = groups.keys.toList()..sort();
    return Drawer(
      child: SafeArea(
        child: Column(
          children: [
            _buildHeader(),
            Expanded(
              child: ListView.builder(
                itemCount: sortedKeys.length,
                itemBuilder: (ctx, i) {
                  final chapterIdx = sortedKeys[i];
                  final items = groups[chapterIdx]!
                    ..sort((a, b) => b.updatedAt.compareTo(a.updatedAt));
                  return Column(
                    crossAxisAlignment: CrossAxisAlignment.start,
                    children: [
                      ListTile(
                        title: Text(chapterTitles[chapterIdx] ?? '第 ${chapterIdx + 1} 章',
                            style: const TextStyle(fontWeight: FontWeight.bold)),
                      ),
                      ...items.map((a) => AnnotationTile(
                            annotation: a,
                            chapterTitle: null,
                            onTap: () => onTap(a.id),
                            onLongPress: () => onLongPress(a.id),
                          )),
                    ],
                  );
                },
              ),
            ),
          ],
        ),
      ),
    );
  }

  Widget _buildHeader() => ListTile(
        leading: const Icon(Icons.menu_book),
        title: const Text('本书高亮与笔记'),
        trailing: IconButton(
          icon: const Icon(Icons.close),
          onPressed: onClose,
        ),
      );
}

// app/lib/features/reader/providers/notes_controller.dart
import 'dart:async';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:uuid/uuid.dart';
import '../../../core/locator.dart';
import '../../../data/models/annotation.dart';
import '../../../data/models/book_note.dart';
import '../../../data/repositories/notes_repository.dart';

class NotesState {
  final BookNote? bookNote;
  final List<Annotation> annotations;
  final bool isLoading;
  final String? error;
  const NotesState({
    this.bookNote,
    this.annotations = const [],
    this.isLoading = false,
    this.error,
  });
  NotesState copyWith({
    BookNote? bookNote,
    List<Annotation>? annotations,
    bool? isLoading,
    String? error,
  }) =>
      NotesState(
        bookNote: bookNote ?? this.bookNote,
        annotations: annotations ?? this.annotations,
        isLoading: isLoading ?? this.isLoading,
        error: error,
      );
}

final currentBookIdProvider = Provider<String?>((_) => null);

final notesRepositoryProvider = Provider<NotesRepository>((ref) {
  throw UnimplementedError('override in main.dart with LocalNotesRepository');
});

final notesControllerProvider =
    StateNotifierProvider.autoDispose<NotesController, AsyncValue<NotesState>>(
  (ref) {
    final bookId = ref.watch(currentBookIdProvider);
    final repo = ref.watch(notesRepositoryProvider);
    return NotesController(bookId: bookId, repository: repo).._init();
  },
);

class NotesController extends StateNotifier<AsyncValue<NotesState>> {
  final String? bookId;
  final NotesRepository repository;
  StreamSubscription<List<Annotation>>? _sub;
  final _uuid = const Uuid();

  NotesController({required this.bookId, required this.repository})
      : super(const AsyncValue.loading());

  Future<void> _init() async {
    if (bookId == null) {
      state = const AsyncValue.data(NotesState());
      return;
    }
    try {
      final note = await repository.getNoteByBook(bookId!);
      _sub = repository.watchAnnotations(bookId!).listen((annotations) {
        state = AsyncValue.data((state.value ?? const NotesState()).copyWith(
          bookNote: note,
          annotations: annotations,
          isLoading: false,
        ));
      });
    } catch (e, st) {
      state = AsyncValue.error(e, st);
    }
  }

  Future<void> addHighlight(
      EpubRangeLocator locator, HighlightColor color, String quote) async {
    final note = await _ensureBookNote();
    final now = DateTime.now();
    await repository.addAnnotation(Annotation(
      id: _uuid.v4(),
      bookNoteId: note.id,
      locator: locator,
      quote: quote,
      note: null,
      color: color,
      tags: const [],
      source: AnnotationSource.manual,
      createdAt: now,
      updatedAt: now,
    ));
  }

  Future<void> updateAnnotation(Annotation annotation) async {
    await repository.updateAnnotation(
        annotation.copyWith(updatedAt: DateTime.now()));
  }

  Future<void> deleteAnnotation(String id) async {
    await repository.deleteAnnotation(id);
  }

  Future<BookNote> _ensureBookNote() async {
    final existing = await repository.getNoteByBook(bookId!);
    if (existing != null) return existing;
    return repository.createEmptyNote(
      id: _uuid.v4(),
      bookId: bookId!,
      bookTitle: '', // caller 提供，M3.2 从 BookRepo 拿
    );
  }

  @override
  void dispose() {
    _sub?.cancel();
    super.dispose();
  }
}
